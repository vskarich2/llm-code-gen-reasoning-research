"""Simple rate limiter."""

from typing import Tuple


def is_rate_limited(count: int, limit: int) -> bool:
    """Return True if the caller should be blocked.

    Args:
        count: number of requests already made (must be a non-negative integer).
        limit: maximum allowed requests (must be a positive integer).

    Returns:
        True if the caller should be blocked (count >= limit), False otherwise.
    """
    if not isinstance(count, int) or not isinstance(limit, int):
        raise TypeError("count and limit must be integers")
    if count < 0:
        raise ValueError("count must be non-negative")
    if limit <= 0:
        raise ValueError("limit must be positive")

    # Block when the number of already-made requests has reached or exceeded the limit.
    return count >= limit


def check_and_increment(current_count: int, limit: int) -> Tuple[bool, int]:
    """Check rate limit and return (blocked, new_count).

    If not blocked, increments the count and returns (False, current_count + 1).
    If blocked, returns (True, current_count) and does not change the count.
    """
    blocked = is_rate_limited(current_count, limit)
    if blocked:
        return True, current_count
    return False, current_count + 1
