from typing import Sequence, List, Dict, Any


def compute_raw_stats(data: Sequence[float]) -> Dict[str, float]:
    """Compute statistics on raw (untransformed) numeric data.

    Returns a dict with keys 'raw_max', 'raw_min', and 'raw_sum'. For empty
    or falsy inputs a consistent zero summary is returned.
    """
    if not data:
        return {"raw_max": 0, "raw_min": 0, "raw_sum": 0}
    return {"raw_max": max(data), "raw_min": min(data), "raw_sum": sum(data)}


def normalize(data: Sequence[float]) -> List[float]:
    """Normalize numeric data into the 0..1 range.

    If the input is empty or falsy an empty list is returned. If all values are
    identical, the function returns a list with the midpoint value 0.5 for each
    element to avoid division by zero while providing a deterministic output.
    """
    if not data:
        return []
    lo, hi = min(data), max(data)
    if hi == lo:
        return [0.5] * len(data)
    return [(x - lo) / (hi - lo) for x in data]


def pipeline(data: Sequence[float]) -> Dict[str, Any]:
    """Run the metrics pipeline.

    The pipeline returns a dict containing:
      - 'raw_stats': statistics computed from the original input data
      - 'cleaned': the normalized version of the input data

    It is critical that raw_stats are computed from the original input (not
    from the normalized values)."""
    # Compute raw statistics from the original data before any transformation.
    raw_stats = compute_raw_stats(data)

    # Produce the normalized/cleaned representation separately.
    cleaned = normalize(data)

    return {"raw_stats": raw_stats, "cleaned": cleaned}


def format_report(result: Dict[str, Any]) -> str:
