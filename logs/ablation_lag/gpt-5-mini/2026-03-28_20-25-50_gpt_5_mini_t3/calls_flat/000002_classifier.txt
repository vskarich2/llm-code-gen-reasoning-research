These changes ensure create_config returns an independent dict and reset_defaults reliably restores the original defaults.

_INITIAL_DEFAULTS = {"timeout": 30, "retries": 3, "debug": False}
# Module-level defaults that callers can inspect but should not mutate directly

def create_config(overrides=None):
    """Return a new config dict based on current DEFAULTS, with optional overrides.

    The returned dict is a shallow copy of DEFAULTS so modifying the returned
    dict will not change the module DEFAULTS.
    """
    config = DEFAULTS.copy()
    if overrides:
        config.update(overrides)
    return config


def reset_defaults():
    """Restore module DEFAULTS to the factory values from _INITIAL_DEFAULTS."""
    global DEFAULTS
    DEFAULTS = _INITIAL_DEFAULTS.copy()
