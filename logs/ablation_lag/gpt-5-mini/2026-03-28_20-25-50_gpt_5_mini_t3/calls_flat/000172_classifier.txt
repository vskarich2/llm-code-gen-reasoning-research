"""Feature flag system with snake_case keys."""

import re

FLAGS = {
    "dark_mode": True,
    "beta_features": False,
    "new_dashboard": True,
    "analytics_v2": False,
}


def _to_snake_case(name):
    """Convert camelCase or mixed-case names to snake_case.

    This helper accepts a string and returns a lowercased underscore-separated
    version. It also replaces dashes with underscores.
    """
    if not isinstance(name, str):
        raise TypeError("flag_name must be a string")
    # Insert underscores between lower->Upper and between acronym/word boundaries
    s1 = re.sub(r'(.)([A-Z][a-z]+)', r'\1_\2', name)
    s2 = re.sub(r'([a-z0-9])([A-Z])', r'\1_\2', s1)
    return s2.replace('-', '_').lower()


def is_enabled(flag_name):
    """Check if a feature flag is enabled.

    Accepts snake_case or camelCase flag names. We prefer an exact key match first
    (preserving explicit snake_case lookups), then try a normalized snake_case
    form if the exact key isn't present. If the flag still isn't found we fall
    back to False (preserving previous behavior for unknown flags).

    Args:
        flag_name: the flag key to look up (snake_case or camelCase)

    Returns:
        True if the flag is enabled, False otherwise.
    """
    # Prefer exact match first to avoid surprising collisions
    if flag_name in FLAGS:
        return bool(FLAGS[flag_name])

    normalized = _to_snake_case(flag_name)
    return bool(FLAGS.get(normalized, False))


def list_flags():
    """Return all flag names."""
    return list(FLAGS.keys())
