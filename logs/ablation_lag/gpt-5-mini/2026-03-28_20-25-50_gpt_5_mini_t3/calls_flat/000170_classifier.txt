```python
 """Feature flag system with snake_case keys.

This module stores feature flags in FLAGS using snake_case keys. Callers may pass
flag names in different common forms (snake_case, camelCase, or kebab-case); the
API normalizes those inputs to snake_case before performing lookups. Unknown
flags raise a KeyError so callers cannot be misled by a silent default value.
"""

import re

FLAGS = {
    "dark_mode": True,
    "beta_features": False,
    "new_dashboard": True,
    "analytics_v2": False,
}


def _normalize_flag_name(flag_name):
    """Normalize a flag name to snake_case.

    Accepts snake_case, camelCase, and kebab-case. Returns a snake_case string.
    """
    if not isinstance(flag_name, str):
        raise TypeError("flag_name must be a str")

    s = flag_name.strip()
    # convert hyphens to underscores (kebab-case -> snake_case)
    s = s.replace("-", "_")
    # Convert CamelCase or camelCase to snake_case using regex.
    # First insert underscore between a lower->Upper transition, then lower all.
    s = re.sub(r"(.)([A-Z][a-z]+)", r"\1_\2", s)
    s = re.sub(r"([a-z0-9])([A-Z])", r"\1_\2", s)
    return s.lower()


def is_enabled(flag_name):
    """Check if a feature flag is enabled.

    Args:
        flag_name: the flag key to look up (accepts snake_case, camelCase, or kebab-case)

    Returns:
        True if the flag is enabled, False otherwise.

    Raises:
        KeyError: if the (normalized) flag name is not present in FLAGS.
        TypeError: if flag_name is not a string.
    """
    normalized = _normalize_flag_name(flag_name)
    if normalized not in FLAGS:
        raise KeyError(
            f"Unknown feature flag: {flag_name!r} (normalized to {normalized!r})."
        )
    return bool(FLAGS[normalized])


def list_flags():
    """Return all flag names (snake_case) known to the system."""
    return list(FLAGS.keys())

```
