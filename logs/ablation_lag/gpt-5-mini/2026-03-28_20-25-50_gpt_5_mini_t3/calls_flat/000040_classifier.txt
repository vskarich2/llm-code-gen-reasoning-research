Settings are stored in a module-private _settings dict. Avoid capturing values at import
time so callers always observe the current configuration when they call accessors.
"""

_settings = {"host": "localhost", "port": 8080, "debug": False}


def get_host():
    """Return the current host setting (read lazily from _settings)."""
    return _settings["host"]


def get_settings():
    """Return a shallow copy of the current settings dict."""
    return dict(_settings)


def reset_settings():
    """Reset settings to default values (mutate existing dict to preserve references)."""
    _settings.clear()
    _settings.update({"host": "localhost", "port": 8080, "debug": False})


def configure(host=None, port=None, debug=None):
    """Update settings in place."""
    if host is not None:
        _settings["host"] = host
    if port is not None:
        _settings["port"] = port
    if debug is not None:
        _settings["debug"] = debug
