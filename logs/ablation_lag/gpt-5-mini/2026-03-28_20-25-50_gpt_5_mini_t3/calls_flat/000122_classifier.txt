```python
"""Access control: map roles to permission sets."""

# Explicit mapping of roles to permissions. Ensure all expected roles are present
# so lookups don't silently fall back to an empty set.
ROLE_PERMISSIONS = {
    "admin": {"read", "write", "delete", "manage_users"},
    "moderator": {"read", "delete", "ban_users"},
    "user": {"read", "write"},
}

# Convenience constant for callers that need to know valid role names.
ROLE_NAMES = frozenset(ROLE_PERMISSIONS.keys())


def get_permissions(role):
    """Return the set of permissions for the given role.

    Returns a copy of the permission set stored in ROLE_PERMISSIONS so callers
    cannot accidentally mutate the module-level mapping. If the role is
    unknown, return an empty set (preserves previous behavior).
    """
    perms = ROLE_PERMISSIONS.get(role)
    if perms is None:
        return set()
    return set(perms)


def has_permission(role, action):
    """Check whether a role has a specific permission."""
    return action in get_permissions(role)

```
