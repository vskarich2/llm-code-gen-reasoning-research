```python
import warnings

"""Access control: map roles to permission sets."""

ROLE_PERMISSIONS = {
    "admin": {"read", "write", "delete", "manage_users"},
    "user": {"read", "write"},
    # moderator role: can read and remove problematic content but not manage users
    "moderator": {"read", "delete"},
}


def get_permissions(role):
    """Return the set of permissions for the given role.

    If the role is unknown, emit a warning and return an empty set so callers
    are alerted to potential configuration or spelling errors while preserving
    backward-compatible behavior.
    """
    if role not in ROLE_PERMISSIONS:
        warnings.warn(f"Unknown role '{role}'; treating permissions as empty set", UserWarning)
        return set()
    return ROLE_PERMISSIONS[role]


def has_permission(role, action):
    """Check whether a role has a specific permission."""
    return action in get_permissions(role)

```
