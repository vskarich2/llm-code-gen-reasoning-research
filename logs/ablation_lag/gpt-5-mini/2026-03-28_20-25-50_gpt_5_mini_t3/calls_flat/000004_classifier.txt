import copy

DEFAULTS = {"timeout": 30, "retries": 3, "debug": False}


def create_config(overrides=None):
    """Return a config dict based on defaults, with optional overrides.

    Always returns a fresh copy so callers cannot mutate the module-level
    DEFAULTS by accident. Accepts an optional dict of overrides.
    """
    config = copy.deepcopy(DEFAULTS)
    if overrides:
        if not isinstance(overrides, dict):
            raise TypeError("overrides must be a dict")
        config.update(copy.deepcopy(overrides))
    return config


def reset_defaults():
    """Restore defaults to factory values in-place (preserves object identity).

    Modifying the dict in-place ensures any external references to the
    DEFAULTS object observe the reset.
    """
    DEFAULTS.clear()
    DEFAULTS.update({"timeout": 30, "retries": 3, "debug": False})
