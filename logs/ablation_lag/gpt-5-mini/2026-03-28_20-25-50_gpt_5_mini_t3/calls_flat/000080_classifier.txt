_status = "idle"

def reset():
    global _status, _data
    _status = "idle"

def load(source):
    """Load data from source. Sets status on success only."""
    global _status, _data

    # Normalize source into a list when possible so we can determine
    # whether it is empty and return a stable data structure.
    if source is None:
        _data = []
        _status = "empty"
        return _data

    try:
        items = [x for x in source]
    except TypeError:
        # Non-iterable source: treat as empty to avoid crashing.
        _data = []
        _status = "empty"
        return _data

    _data = items
    _status = "loaded" if len(items) > 0 else "empty"
    return _data
