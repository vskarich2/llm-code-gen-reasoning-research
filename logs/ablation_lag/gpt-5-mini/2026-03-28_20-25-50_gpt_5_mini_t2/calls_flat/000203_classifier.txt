    to_remove = [k for k in _store if k.startswith(prefix)]

def sync_user_snapshot(user, overwrite=False):
    """Store a user's display name in the cache.

    if the key is absent.

    if overwrite:
        cache_put(key, user["name"])
    else:
        cache_put_if_absent(key, user["name"])
