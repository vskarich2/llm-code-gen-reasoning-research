```python
import re

"""Feature flag system with snake_case keys.

This module accepts common naming variants (camelCase, PascalCase, kebab-case,
space separated) for flag lookups by normalizing them to snake_case before
checking the FLAGS mapping.
"""

FLAGS = {
    "dark_mode": True,
    "beta_features": False,
    "new_dashboard": True,
    "analytics_v2": False,
}


def _normalize_flag_name(name):
    """Normalize a flag name to snake_case.

    Accepts camelCase, PascalCase, kebab-case, and space-separated names.
    Raises TypeError if name is not a string.
    """
    if not isinstance(name, str):
        raise TypeError("flag_name must be a string")

    # Replace common separators with underscores
    s = name.replace('-', '_').replace(' ', '_')

    # Insert underscores between lower->Upper transitions and between letter-digit/ digit-letter where helpful
    s = re.sub(r'(.)([A-Z][a-z]+)', r'\1_\2', s)
    s = re.sub(r'([a-z0-9])([A-Z])', r'\1_\2', s)

    # Collapse multiple underscores and lower-case the result
    s = re.sub(r'_+', '_', s).lower()

    return s


def is_enabled(flag_name):
    """Check if a feature flag is enabled.

    The provided flag_name is normalized to snake_case before lookup, so callers
    can pass common variants like 'darkMode', 'DarkMode', 'dark-mode', or
    'dark mode'. Non-string inputs raise TypeError. Unknown (but validly-typed)
    flag names return False.

    Args:
        flag_name: the flag key to look up (string; common naming variants allowed)

    Returns:
        True if the flag is enabled, False otherwise.
    """
    normalized = _normalize_flag_name(flag_name)
    if normalized in FLAGS:
        return bool(FLAGS[normalized])
    return False


def list_flags():
    """Return all defined flag names (in their canonical snake_case form)."""
    return list(FLAGS.keys())

```
