from collections.abc import Mapping
import copy

DEFAULT_TEMPLATE = {"timeout": 30, "retries": 3, "debug": False}
DEFAULTS = copy.deepcopy(DEFAULT_TEMPLATE)


def create_config(overrides=None):
    """Return a new config dict based on defaults, with optional overrides.

    The returned dict is a deep copy of the default template so callers may
    modify it without affecting module-level DEFAULTS. If overrides is
    provided it must be a mapping; its keys will override the top-level
    keys in the returned config.
    """
    config = copy.deepcopy(DEFAULTS)
    if overrides is None:
        return config
    if not isinstance(overrides, Mapping):
        raise TypeError("overrides must be a mapping or None")
    config.update(overrides)
    return config


def reset_defaults():
    """Restore module DEFAULTS to the original factory values."""
    global DEFAULTS
    DEFAULTS = copy.deepcopy(DEFAULT_TEMPLATE)
