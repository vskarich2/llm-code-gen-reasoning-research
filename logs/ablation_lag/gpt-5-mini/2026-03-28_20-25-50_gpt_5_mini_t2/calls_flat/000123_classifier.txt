```python
"""Access control: map roles to permission sets."""

# Define permissions for each role. Keys are normalized (lowercase).
ROLE_PERMISSIONS = {
    "admin": {"read", "write", "delete", "manage_users"},
    "user": {"read", "write"},
    "moderator": {"read", "delete", "manage_reports"},
}


def _normalize_role(role):
    """Return a normalized role string or None for non-string inputs."""
    if not isinstance(role, str):
        return None
    return role.strip().lower()


def _normalize_action(action):
    """Return a normalized action string or None for non-string inputs."""
    if not isinstance(action, str):
        return None
    return action.strip().lower()


def get_permissions(role):
    """Return the set of permissions for the given role.

    Role input is normalized (strip + lowercase). If the role is unknown or not a string,
    an empty set is returned.
    """
    normalized = _normalize_role(role)
    if normalized is None:
        return set()
    # return a copy to prevent callers from mutating the stored sets
    return ROLE_PERMISSIONS.get(normalized, set()).copy()


def has_permission(role, action):
    """Check whether a role has a specific permission.

    Both role and action are normalized for robust lookups.
    Non-string actions return False.
    """
    normalized_action = _normalize_action(action)
    if normalized_action is None:
        return False
    return normalized_action in get_permissions(role)

```
