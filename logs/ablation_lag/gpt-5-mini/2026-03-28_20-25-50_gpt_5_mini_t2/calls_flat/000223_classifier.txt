 """Bank with non-atomic check-then-act withdrawal, simulated via steps."""

_accounts = {}


def reset():
    _accounts.clear()


def create_account(name, balance):
    _accounts[name] = balance


def get_balance(name):
    return _accounts.get(name, 0)


def check_balance(name, amount):
    """CHECK: is balance sufficient?"""
    return _accounts.get(name, 0) >= amount


def do_withdraw(name, amount):
    """ACT: decrement balance. Assumes caller verified."""
    _accounts[name] = _accounts.get(name, 0) - amount


def reserve_withdraw(name, amount):
    """Atomically check-and-decrement (reserve funds) if sufficient.

    Returns True if reservation (withdrawal) succeeded, False otherwise.
    This prevents separate check and act steps from both succeeding under interleaving.
    """
    if _accounts.get(name, 0) >= amount:
        _accounts[name] = _accounts.get(name, 0) - amount
        return True
    return False


def run_steps(steps):
    results = []
    for fn, args in steps:
        results.append(fn(*args))
    return results


def make_withdraw_steps(name, amount):
    """Split withdrawal into separate check and act steps.

    FIX: perform an atomic reservation during the check step so two checks
    cannot both succeed before either debits the account. The act step
    becomes a no-op that simply reports the previously recorded outcome.
    """
    result = {"approved": False}

    def step_check():
        # Atomically attempt to reserve (withdraw) the funds now.
        result["approved"] = reserve_withdraw(name, amount)
        return ("check", result["approved"])

    def step_act():
        # Act is now a no-op because the check already performed the
        # reservation/debit if it was approved.
        return ("act", result["approved"])

    return step_check, step_act


def sequential_withdrawals():
    """Two withdrawals of 80 from balance=100. Sequential: second denied."""
    reset()

    check_a, act_a = make_withdraw_steps("alice", 80)
