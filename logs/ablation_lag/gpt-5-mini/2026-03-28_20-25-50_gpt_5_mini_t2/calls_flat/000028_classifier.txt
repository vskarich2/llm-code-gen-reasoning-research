```python
"""Product catalog with caching and cache-invalidation on updates.

This module stores products in an in-memory _db and keeps a per-id cache
in _cache. update_product now invalidates the cache entry for the updated
product so get_product does not return stale data. Access is protected by
an RLock to avoid simple race conditions when used from multiple threads.
"""

import threading

_db = {}
_cache = {}
_lock = threading.RLock()


def add_product(product_id, name, price):
    """Add a product to the database.

    If a cached entry exists for the same product_id, remove it so future
    lookups reflect the newly added product.
    """
    with _lock:
        _db[product_id] = {"name": name, "price": price}
        _cache.pop(product_id, None)


def get_product(product_id):
    """Get product by ID, using cache for repeated lookups.

    Returns a copy of the cached product dict (or None if not found).
    """
    with _lock:
        if product_id in _cache:
            return _cache[product_id]
        product = _db.get(product_id)
        if product is not None:
            # store a shallow copy to avoid callers mutating the stored db object
            _cache[product_id] = dict(product)
        return _cache.get(product_id)


def update_product(product_id, **fields):
    """Update product fields in the database.

    Invariant: get_product() must return current data after update.
    This function updates the DB and invalidates the cache entry for the
    product so subsequent get_product calls rebuild the cache.
    """
    with _lock:
        if product_id not in _db:
            return False
        _db[product_id].update(fields)
        # Invalidate the cached copy so get_product returns fresh data
        _cache.pop(product_id, None)
        return True


def reset():
    """Clear all data."""
    with _lock:
        _db.clear()
        _cache.clear()

```
