def update_profile(user, changes):
    """Update user profile fields from changes dict.

    Invariant: display_name must always equal name.
    """
    # If an explicit name change is provided, prefer it and sync display_name.
    if "name" in changes:
        new_name = changes["name"]
        user["name"] = new_name
        user["display_name"] = new_name
    # If name wasn't provided but display_name was, treat it as a name change
    # and sync both fields so the invariant holds.
    elif "display_name" in changes:
        new_name = changes["display_name"]
        user["name"] = new_name
        user["display_name"] = new_name

    # Apply other changes; skip name/display_name because already handled.
    for key, value in changes.items():
        if key in ("name", "display_name"):
            continue

        elif key == "age":
            user["age"] = value
    return user
