from cache import put_local, put_shared, invalidate_shared, invalidate_local

_db = {}


def add_product(product_id, name, price):
    """Add product to database."""
    _db[product_id] = {"name": name, "price": price}


def db_get(product_id):
    """Raw DB lookup."""
    p = _db.get(product_id)
    return dict(p) if p else None


def update_product(product_id, **fields):
    """Update product in DB and invalidate caches.

    Invariant: after update, all cache layers must be invalidated.
    """
    if product_id not in _db:
        return False
    _db[product_id].update(fields)

    # Invalidate local cache in this process so api.get_product won't return stale data.
    invalidate_local(product_id)

    # Proactively update the shared cache with the new value so other readers
    # that consult shared will see the updated product. This reduces the window
    # where other processes might observe stale data in shared cache.
    updated = dict(_db[product_id])
    put_shared(product_id, updated)

    # Note: cross-process/local-cache invalidation requires an external
    # mechanism (pub/sub, TTL, etc.). This change addresses the immediate bug
    # where the current process' local cache was not invalidated.
    return True


def reset():
    """Clear DB."""
    _db.clear()
