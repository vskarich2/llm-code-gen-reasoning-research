def is_rate_limited(count, limit):
    """Return True when the number of previous requests has reached or exceeded the allowed limit.

    Args:
        count: current number of requests already made (0-based).
        limit: maximum allowed requests.

    Returns:
        True if the caller should be blocked, False if allowed.
    """
    # Block when the count is greater than or equal to the limit.
    return count >= limit


def check_and_increment(current_count, limit):

    """

        return True, current_count
    return False, current_count + 1
