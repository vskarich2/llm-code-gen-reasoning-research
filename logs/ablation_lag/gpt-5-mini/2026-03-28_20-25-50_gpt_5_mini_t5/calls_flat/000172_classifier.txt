"""Feature flag system with snake_case keys."""

import re
import warnings

FLAGS = {
    "dark_mode": True,
    "beta_features": False,
    "new_dashboard": True,
    "analytics_v2": False,
}


def _to_snake(name):
    """Convert common name styles (camelCase, PascalCase, kebab-case) to snake_case.

    This handles typical camel/PascalCase and hyphenated names. If the input is
    already snake_case it will be returned unchanged (modulo lowercasing).
    """
    if not isinstance(name, str):
        raise TypeError("flag_name must be a string")

    # Replace hyphens with underscores first
    s = name.replace('-', '_')
    # Convert CamelCase / camelCase to snake_case
    # First pass: put underscore between lower-to-upper transitions
    s = re.sub(r'(.)([A-Z][a-z]+)', r'\1_\2', s)
    # Second pass: put underscore between a lower/number and upper
    s = re.sub(r'([a-z0-9])([A-Z])', r'\1_\2', s)
    return s.lower()


def is_enabled(flag_name):
    """Check if a feature flag is enabled.

    Args:
        flag_name: the flag key to look up (may be snake_case or camelCase)

    Returns:
        True if the flag is enabled, False otherwise.

    Behavior:
        - If flag_name matches a key in FLAGS exactly, that value is returned.
        - Otherwise the name is normalized to snake_case and looked up.
        - If still not found, a warning is issued and False is returned
          (preserves previous behavior but avoids silent failures).
    """
    if flag_name in FLAGS:
        return FLAGS[flag_name]

    try:
        normalized = _to_snake(flag_name)
    except TypeError:
        raise

    if normalized in FLAGS:
        return FLAGS[normalized]

    warnings.warn(
        "Feature flag '%s' not found; returned False. "
        "Ensure you use snake_case keys (e.g. 'dark_mode')." % (flag_name,),
        UserWarning,
    )
    return False


def list_flags():
    """Return all flag names."""
    return list(FLAGS.keys())
