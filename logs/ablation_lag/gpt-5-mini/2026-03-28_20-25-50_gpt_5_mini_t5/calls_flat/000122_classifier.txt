```python
"""Access control: map roles to permission sets.

This module defines a clear ROLE_PERMISSIONS mapping and helper functions
for retrieving and checking permissions. The mapping now explicitly includes
all domain roles (e.g. 'moderator') so callers of get_permissions and
has_permission do not silently get an empty set for a known role.
"""

from typing import Set

ROLE_PERMISSIONS = {
    "admin": {"read", "write", "delete", "manage_users"},
    "moderator": {"read", "delete"},
    "user": {"read", "write"},
}


def get_permissions(role: str) -> Set[str]:
    """Return the set of permissions for the given role.

    If the role is unknown, return an empty set.
    """
    return ROLE_PERMISSIONS.get(role, set())


def has_permission(role: str, action: str) -> bool:
    """Check whether a role has a specific permission."""
    return action in get_permissions(role)

```
