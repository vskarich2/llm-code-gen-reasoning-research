def reset():

    _sent = []


def send(msg):
    """Send a message. Always succeeds. Appends to _sent."""

    return True


def retry_send(msg, max_retries=2):
    """Send with retry. Attempts to send up to max_retries times.
    Returns True on success, False if all attempts fail.
    Stops retrying immediately after a successful send.
    """
    if max_retries <= 0:
        return False

    for attempt in range(1, max_retries + 1):
        try:
            result = send(msg)
        except Exception:
            result = False

        if result:
            # success — stop retrying
            return True
        # else: failed attempt, try again
    return False
