_store = {}


def cache_put(key, value):
    _store[key] = value


def cache_delete(key):
    _store.pop(key, None)


def cache_put_if_absent(key, value):
    if key not in _store:
        _store[key] = value


def cache_clear_prefix(prefix):
    to_remove = [k for k in _store if k.startswith(prefix)]
    for k in to_remove:
        del _store[k]


def sync_user_to_cache(user, force=False):
    """Write a user's display name to the cache.

    By default (force=False) this will only set the value if absent (preserving
    previous refresh_user_snapshot behavior). If force=True, this will unconditionally
    overwrite the cached value (preserving previous sync_user_to_cache behavior).
    """
    key = f"user:{user['id']}"
    if force:
        cache_put(key, user["name"])
    else:
        cache_put_if_absent(key, user["name"])


# Backwards-compatible wrapper using the default (non-overwrite) behavior
def refresh_user_snapshot(user):
    return sync_user_to_cache(user, force=False)
