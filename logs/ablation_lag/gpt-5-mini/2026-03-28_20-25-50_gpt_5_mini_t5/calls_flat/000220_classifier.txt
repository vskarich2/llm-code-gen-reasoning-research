"""Counter with non-atomic read-modify-write, simulated via step functions."""

_value = 0


def reset():
    global _value
    _value = 0


def get():
    return _value


def _set(v):
    global _value
    _value = v


def run_steps(steps):
    results = []
    for fn, args in steps:
        results.append(fn(*args))
    return results


def make_increment_steps():
    """Provide steps for an increment. The write step performs the read-modify-write
    at the time the write step runs, avoiding use of a stale captured read value.
    """

    def step_read():
        # Return the current value at read-time; not relied on for the write.
        current = get()
        return ("read", current)

    def step_write():
        # Perform the read-modify-write atomically with respect to step scheduling
        # by reading the current value here and then writing the incremented value.
        new = get() + 1
        _set(new)
        return ("write", new)

    return step_read, step_write


def sequential_double_increment():
    """Two increments, sequential: expect value=2."""
    reset()
    read_a, write_a = make_increment_steps()
    read_b, write_b = make_increment_steps()
    run_steps([(read_a, ()), (write_a, ()), (read_b, ()), (write_b, ())])
    return get()


def interleaved_double_increment():
    """Two increments, interleaved. Fixed: each write reads current at write-time so both increments
    are applied correctly regardless of interleaving."""

    read_b, write_b = make_increment_steps()
    run_steps([(read_a, ()), (read_b, ()), (write_a, ()), (write_b, ())])
    return get()
