from models import Account
from ledger import record_debit, record_credit, record_transfer_attempt
from audit import emit_transfer_event, emit_failure_alert


def validate_transfer(sender, amount):
    if amount <= 0:
        raise ValueError("amount must be positive")
    if sender.balance < amount:
        raise ValueError("insufficient funds")


def _core_transfer(sender, receiver, amount):
    """Core business logic: validate and move balances.

    This function performs the debit then attempts the credit. If the credit phase
    fails it rolls back the debit to avoid leaving the system in a partially-applied
    state.
    """
    validate_transfer(sender, amount)

    # Apply debit first
    sender.balance -= amount

    try:
        # Simulate a transient failure during the credit phase
        if random.random() < 0.3:
            raise RuntimeError("transient failure during credit")

        # Apply credit
        receiver.balance += amount

    except Exception:
        # Rollback debit to preserve invariants, then re-raise
        sender.balance += amount
        raise


def execute_transfer(sender, receiver, amount):
    """Observability wrapper around the core transfer.

    Responsibilities:
    - record an initial transfer attempt (pending)
    - invoke the core transfer
    - on success, record debit/credit, mark attempt completed and emit transfer event
    - on failure, mark attempt failed and emit a failure alert

    Observability calls are isolated from the core transfer and are guarded so
    they don't leave the system in an inconsistent state if they themselves fail.
    """
    # Mark attempt pending
    try:
        record_transfer_attempt(sender.account_id, receiver.account_id, amount, status="pending")
    except Exception:
        # If recording the pending attempt fails, continue; do not block the transfer
        pass

    try:
        _core_transfer(sender, receiver, amount)
    except Exception as e:
        # Core transfer failed: attempt to mark failed and emit an alert, then re-raise
        try:
            record_transfer_attempt(sender.account_id, receiver.account_id, amount, status="failed")
        except Exception:
            pass
        try:
            emit_failure_alert(sender.account_id, receiver.account_id, amount, str(e))
        except Exception:
            pass
        raise

    # Core transfer succeeded: record ledger entries and emit event
    try:
        record_debit(sender.account_id, amount)
        record_credit(receiver.account_id, amount)
        try:
            record_transfer_attempt(sender.account_id, receiver.account_id, amount, status="completed")
        except Exception:
            pass
        emit_transfer_event(sender.account_id, receiver.account_id, amount)
    except Exception as obs_exc:
        # Observability failed after a successful transfer. Try to emit an alert, but
        # do not attempt to roll back balances (transfer succeeded).
        try:
            emit_failure_alert(sender.account_id, receiver.account_id, amount, f"observability failure: {obs_exc}")
        except Exception:
            pass
        # Re-raise so callers know something went wrong with side-effects
        raise
