import logging

logger = logging.getLogger(__name__)

ROLE_PERMISSIONS = {
    "admin": {"read", "write", "delete", "manage_users"},
    "user": {"read", "write"},
    # Fixed: explicitly include the "moderator" role so it doesn't fall through to an empty set
    "moderator": {"read", "delete", "manage_reports"},
}


def get_permissions(role):
    """Return the set of permissions for the given role.

    If the role is unknown, log a warning and return an empty set (preserves previous behavior).
    """
    if role not in ROLE_PERMISSIONS:
        logger.warning("Unknown role requested: %r. Returning empty permission set.", role)
        return set()
    return ROLE_PERMISSIONS[role]


def has_permission(role, action):
    """Check whether a role has a specific permission."""
    return action in get_permissions(role)
