## Failure Mechanism

Since 'darkMode' is not a key in FLAGS, .get returns the default False, so the caller receives False. The causal chain is: caller supplies camelCase -> is_enabled doesn't normalize the name -> dictionary lookup misses -> .get returns the silent default False -> caller gets an incorrect False value.

```python
"""Feature flag system with snake_case keys.

This module accepts both snake_case and camelCase flag names when checking
flags. It normalizes incoming names to snake_case and performs an explicit
lookup so that unknown flags raise KeyError instead of silently returning
False.
"""

import re

FLAGS = {
    "dark_mode": True,
    "beta_features": False,
    "new_dashboard": True,
    "analytics_v2": False,
}


def _to_snake(flag_name):
    """Normalize a flag name to snake_case.

    Accepts either snake_case (unchanged) or camelCase (converted).
    Raises TypeError if flag_name is not a string.
    """
    if not isinstance(flag_name, str):
        raise TypeError("flag_name must be a string")

    # Fast path: already looks like snake_case (lowercase with underscores)
    if flag_name == flag_name.lower() and "_" in flag_name:
        return flag_name

    # Convert camelCase or PascalCase to snake_case: e.g. "darkMode" -> "dark_mode"
    s = re.sub(r"([A-Z])", r"_\1", flag_name).lower()
    # Collapse any accidental leading/trailing underscores
    s = s.strip("_")
    return s


def is_enabled(flag_name):
    """Check if a feature flag is enabled.

    Args:
        flag_name: the flag key to look up. You may pass snake_case or camelCase.

    Returns:
        True if the flag is enabled, False otherwise.

    Raises:
        KeyError: if the normalized flag name does not exist in FLAGS.
        TypeError: if flag_name is not a string.
    """
    normalized = _to_snake(flag_name)

    # Use direct indexing so missing flags raise KeyError instead of silently
    # returning a default. This prevents callers from mistakenly believing a
    # feature is disabled when they misspelled or used the wrong case.
    try:
        return FLAGS[normalized]
    except KeyError as exc:
        available = ", ".join(sorted(FLAGS.keys()))
        raise KeyError(
            f"Unknown feature flag '{flag_name}' (normalized to '{normalized}'). "
            f"Available flags: {available}"
        ) from exc


def list_flags():
    """Return all flag names (canonical snake_case)."""
    return sorted(FLAGS.keys())

```
