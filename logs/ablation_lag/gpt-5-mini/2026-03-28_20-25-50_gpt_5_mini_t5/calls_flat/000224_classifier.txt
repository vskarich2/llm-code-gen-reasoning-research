import threading

"""Bank with safe check-then-act withdrawal, simulated via steps.

This version prevents the interleaving bug by making the ACT operation
perform an authoritative check-and-act under a per-account lock.
"""

_accounts = {}
_locks = {}


def reset():
    _accounts.clear()
    _locks.clear()


def create_account(name, balance):
    _accounts[name] = balance
    # ensure a lock exists for the account
    _locks[name] = threading.Lock()


def get_balance(name):
    lock = _locks.get(name)
    if lock:
        with lock:
            return _accounts.get(name, 0)
    return _accounts.get(name, 0)


def check_balance(name, amount):
    """CHECK: is balance sufficient? (best-effort view)

    Note: this is a non-authoritative check; the authoritative check-and-act
    happens in do_withdraw which is locked.
    """
    lock = _locks.get(name)
    if lock:
        with lock:
            return _accounts.get(name, 0) >= amount
    return _accounts.get(name, 0) >= amount


def do_withdraw(name, amount):
    """ACT: atomically verify balance and decrement under a lock.

    Returns True if withdrawal succeeded, False otherwise.
    """
    # Ensure a lock exists for this account (on-demand)
    lock = _locks.setdefault(name, threading.Lock())
    with lock:
        current = _accounts.get(name, 0)
        if current >= amount:
            _accounts[name] = current - amount
            return True
        return False


def run_steps(steps):
    results = []
    for fn, args in steps:
        results.append(fn(*args))
    return results


def make_withdraw_steps(name, amount):
    """Split withdrawal into separate check and act steps.

    The CHECK step still provides a preliminary approval, but the ACT step
    performs the authoritative, locked check-and-act by calling do_withdraw.
    This prevents two interleaved checks from both leading to successful
    debits.
    """
    result = {"approved": False}

    def step_check():
        result["approved"] = check_balance(name, amount)
        return ("check", result["approved"])

    def step_act():
        # Perform the authoritative, locked check-and-act. Update the
        # approval to reflect the actual outcome at the time of the act.
        result["approved"] = do_withdraw(name, amount)
        return ("act", result["approved"])

    return step_check, step_act


def sequential_withdrawals():
    """Two withdrawals of 80 from balance=100. Sequential: second denied."""
    reset()
    create_account("alice", 100)
    check_a, act_a = make_withdraw_steps("alice", 80)
    check_b, act_b = make_withdraw_steps("alice", 80)
    run_steps([(check_a, ()), (act_a, ()), (check_b, ()), (act_b, ())])
    return get_balance("alice")


def interleaved_withdrawals():
    """Two withdrawals of 80, interleaved: fixed — only one approved."""
    reset()
    create_account("alice", 100)
    check_a, act_a = make_withdraw_steps("alice", 80)
    check_b, act_b = make_withdraw_steps("alice", 80)
    run_steps([(check_a, ()), (check_b, ()), (act_a, ()), (act_b, ())])
    return get_balance("alice")
