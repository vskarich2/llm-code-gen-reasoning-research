```python
import re
import warnings

"""Feature flag system with snake_case keys.

This module normalizes incoming flag names to snake_case so callers
can use camelCase or snake_case. Unknown flags still return False,
but a warning is emitted to make unexpected lookups visible.
"""

FLAGS = {
    "dark_mode": True,
    "beta_features": False,
    "new_dashboard": True,
    "analytics_v2": False,
}


def _to_snake_case(name):
    """Convert camelCase or PascalCase to snake_case.

    Leaves an already snake_case string unchanged.
    Raises TypeError for non-string inputs.
    """
    if not isinstance(name, str):
        raise TypeError("flag_name must be a string")
    # Fast path: if it already contains underscores and is lowercase, assume snake_case
    if name == name.lower() and "_" in name:
        return name
    # Insert underscores before capital letters and lower the whole string
    s1 = re.sub(r"(.)([A-Z][a-z]+)", r"\1_\2", name)
    snake = re.sub(r"([a-z0-9])([A-Z])", r"\1_\2", s1).lower()
    return snake


def is_enabled(flag_name):
    """Check if a feature flag is enabled.

    Accepts snake_case or camelCase flag_name. If the flag is unknown,
    emits a warning and returns False (preserves previous behavior but
    surfaces unexpected lookups).

    Args:
        flag_name: the flag key to look up (snake_case or camelCase)

    Returns:
        True if the flag is enabled, False otherwise.
    """
    normalized = _to_snake_case(flag_name)

    # Prefer the normalized key if it exists; also accept the original if it was snake_case
    if normalized in FLAGS:
        return FLAGS[normalized]
    if flag_name in FLAGS:
        return FLAGS[flag_name]

    warnings.warn(
        "Unknown feature flag '%s' (normalized '%s'). Returning False. "
        "Known flags: %s" % (flag_name, normalized, sorted(FLAGS.keys())),
        stacklevel=2,
    )
    return False


def list_flags():
    """Return all flag names (snake_case)."""
    return list(FLAGS.keys())

```
