```python
"""Configuration utilities for alias_config_a.

Provides a DEFAULTS mapping and helpers that avoid accidental aliasing of
module-level state by always copying mutable dicts before returning them.
"""

_DEFAULT_TEMPLATE = {"timeout": 30, "retries": 3, "debug": False}

# Public DEFAULTS for external inspection; can be reset with reset_defaults().
DEFAULTS = dict(_DEFAULT_TEMPLATE)


def create_config(overrides=None):
    """Return a fresh config dict based on DEFAULTS, with optional overrides.

    The returned dict is a copy and mutating it will not affect the module-level
    DEFAULTS mapping.
    """
    config = dict(DEFAULTS)
    if overrides:
        config.update(overrides)
    return config


def reset_defaults():
    """Restore DEFAULTS to the original factory values."""
    global DEFAULTS
    DEFAULTS = dict(_DEFAULT_TEMPLATE)

```
