```python
"""Access control: map roles to permission sets."""

# Centralized role -> permissions mapping (keys are lowercase)
ROLE_PERMISSIONS = {
    "admin": {"read", "write", "delete", "manage_users"},
    "moderator": {"read", "delete", "manage_reports"},
    "user": {"read", "write"},
}

# Common aliases mapped to canonical role keys
ROLE_ALIASES = {
    "mod": "moderator",
}


def get_permissions(role):
    """Return the set of permissions for the given role.

    Role lookup is normalized to lowercase and supports a small set of aliases.
    Unknown roles return an empty set.
    """
    if role is None:
        return set()
    key = str(role).lower()
    key = ROLE_ALIASES.get(key, key)
    return ROLE_PERMISSIONS.get(key, set())


def has_permission(role, action):
    """Check whether a role has a specific permission."""
    return action in get_permissions(role)

```
