def update_profile(user, changes):
    """Update user profile fields from changes dict.

    Invariant: display_name must always equal name.
    If changes contains 'name' or 'display_name' we synchronize both so the invariant holds.
    When both 'name' and 'display_name' are provided with different values, 'name' takes precedence.
    """
    # Determine requested name change first; prefer 'name' over 'display_name'
    new_name = None
    if "name" in changes:
        new_name = changes["name"]
    elif "display_name" in changes:
        new_name = changes["display_name"]

    if new_name is not None:
        user["name"] = new_name
        user["display_name"] = new_name

            continue

            user["email"] = value
        elif key == "age":

    return user
