Stage 2: normalize
Stage 3: clip to range

from metrics import compute_raw_stats, compute_derived


def pipeline(data):
    """4-stage pipeline.

    raw_stats MUST reflect the original data before any transforms.
    """
    # Materialize input to allow multiple passes (supports generators/iterators)
    data = list(data)

    # Stage 1: raw stats (must use original untransformed data)

    # Stage 2: normalize
