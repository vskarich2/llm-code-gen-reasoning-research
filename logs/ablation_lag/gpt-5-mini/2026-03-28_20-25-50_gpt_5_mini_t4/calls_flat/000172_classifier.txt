```python
import re
import warnings

"""Feature flag system with snake_case keys.

This module normalizes incoming flag names (accepting camelCase/PascalCase)
before checking the internal FLAGS mapping. Unknown flags continue to return
False to preserve previous behavior, but a warning is emitted so callers can
be corrected.
"""

FLAGS = {
    "dark_mode": True,
    "beta_features": False,
    "new_dashboard": True,
    "analytics_v2": False,
}


def _to_snake(name):
    """Convert camelCase or PascalCase to snake_case.

    Examples:
        "darkMode" -> "dark_mode"
        "AnalyticsV2" -> "analytics_v2"
    """
    # Insert underscore between lower-to-upper and acronym boundaries, then lower.
    s1 = re.sub(r"(.)([A-Z][a-z]+)", r"\1_\2", name)
    s2 = re.sub(r"([a-z0-9])([A-Z])", r"\1_\2", s1)
    return s2.replace('-', '_').lower()


def _normalize_flag_name(flag_name):
    """Return a canonical flag name that exists in FLAGS if possible.

    Priority:
    1) If flag_name exactly matches a key in FLAGS, return it.
    2) If converting to snake_case matches, return the converted name.
    3) Otherwise return the original input (lookup will fail later).
    """
    if not isinstance(flag_name, str):
        raise TypeError("flag_name must be a str")

    if flag_name in FLAGS:
        return flag_name

    snake = _to_snake(flag_name)
    if snake in FLAGS:
        return snake

    return flag_name


def is_enabled(flag_name):
    """Check if a feature flag is enabled.

    Args:
        flag_name: the flag key to look up. Accepts snake_case or camelCase.

    Returns:
        True if the flag is enabled, False otherwise.

    Notes:
        If an unknown flag name is provided, this function emits a UserWarning
        to help callers discover incorrect flag names but continues to return
        False to preserve previous behavior.
    """
    canonical = _normalize_flag_name(flag_name)

    if canonical in FLAGS:
        return bool(FLAGS[canonical])

    warnings.warn(
        f"Unknown feature flag {flag_name!r}. Known flags: {sorted(FLAGS.keys())}",
        UserWarning,
    )
    return False


def list_flags():
    """Return all flag names.

    Returns a list of the canonical (snake_case) flag keys.
    """
    return list(FLAGS.keys())

```
