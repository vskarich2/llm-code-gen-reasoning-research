DEFAULT_TEMPLATE = {"timeout": 30, "retries": 3, "debug": False}

# Internal module defaults; operations should copy this to avoid exposing
# or mutating the shared structure.
_defaults = DEFAULT_TEMPLATE.copy()


def create_config(overrides=None):
    """Return a fresh config dict based on current defaults, with optional overrides.

    The returned dict is a shallow copy and safe for callers to modify without
    affecting module-level defaults.
    """
    # Make a shallow copy so we never return the internal defaults by reference
    config = _defaults.copy()
    if overrides:
        config.update(overrides)
    return config


def reset_defaults():
    """Restore internal defaults to the original template values."""
    global _defaults
    _defaults = DEFAULT_TEMPLATE.copy()
