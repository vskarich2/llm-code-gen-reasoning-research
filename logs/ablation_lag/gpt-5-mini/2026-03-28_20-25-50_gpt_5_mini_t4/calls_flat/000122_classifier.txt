```python
'''Access control: map roles to permission sets.

Permissions are declared explicitly per role. Permission sets are immutable
(frozenset) for clarity and to avoid accidental mutation. AVAILABLE_ROLES
lists the known roles.
'''

ROLE_PERMISSIONS = {
    "admin": frozenset({"read", "write", "delete", "manage_users"}),
    "moderator": frozenset({"read", "delete"}),
    "user": frozenset({"read", "write"}),
}

AVAILABLE_ROLES = tuple(sorted(ROLE_PERMISSIONS.keys()))


def get_permissions(role):
    """Return the set of permissions for the given role.

    Known roles are listed in AVAILABLE_ROLES. Unknown roles return an empty
    frozenset (no permissions).
    """
    return ROLE_PERMISSIONS.get(role, frozenset())


def has_permission(role, action):
    """Check whether a role has a specific permission."""
    return action in get_permissions(role)

```
