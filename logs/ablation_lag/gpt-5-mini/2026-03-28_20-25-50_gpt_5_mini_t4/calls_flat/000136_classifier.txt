"""Simple rate limiter."""


def is_rate_limited(count, limit):
    """Check if the request count has reached or exceeded the rate limit.

    Args:
        count: non-negative integer representing the current number of requests made
               (0 means no requests yet).
        limit: non-negative integer for the maximum allowed requests.

    Returns:
        True if the caller should be blocked (count >= limit), False if allowed.

    Raises:
        TypeError: if count or limit is not an int.
        ValueError: if count or limit is negative.
    """
    if not isinstance(count, int) or not isinstance(limit, int):
        raise TypeError("count and limit must be integers")
    if count < 0 or limit < 0:
        raise ValueError("count and limit must be non-negative")

    # Block when the recorded count is equal to or greater than the limit.
    # Using '>=' fixes the off-by-one bug where exactly 'limit' requests were
    # previously allowed when they should be blocked.
    return count >= limit


def check_and_increment(current_count, limit):
    """Check rate limit and return (blocked, new_count).

    If not blocked, increments the count by one and returns the new count.

    Args:
        current_count: current number of requests already made (int >= 0).
        limit: maximum allowed requests (int >= 0).

    Returns:
        Tuple (blocked: bool, new_count: int).
    """
    if is_rate_limited(current_count, limit):
        return True, current_count
    return False, current_count + 1
