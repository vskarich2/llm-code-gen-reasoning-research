_default_host = _settings["host"]

    """Return the current host setting."""

    return _settings["host"]

    """Return a shallow copy of the current settings dict."""

    """Reset settings to default values (mutates the existing dict).

    """
    global _default_host
    _settings.clear()

    _default_host = _settings["host"]


def configure(host=None, port=None, debug=None):
    """Update settings.

    Only values that are not None are updated.
    """
    global _default_host
    if host is not None:
        _settings["host"] = host
    if port is not None:
        _settings["port"] = port
    if debug is not None:

    # Keep the module-level attribute in sync for backward compatibility.
