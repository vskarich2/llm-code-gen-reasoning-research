```python
"""Product catalog with caching.

This module stores product data in a simple in-memory database (_db)
and keeps a cache (_cache) of product snapshots for repeated lookups.

Design notes:
- get_product returns a shallow copy of the cached product to avoid
  callers mutating the cached object.
- update_product refreshes the cache entry after updating the database
  so get_product observes current data.
- add_product invalidates any existing cache for the same product_id.
"""

_db = {}
_cache = {}


def add_product(product_id, name, price):
    """Add a product to the database.

    If a cache entry exists for product_id, invalidate it so subsequent
    lookups return fresh data.
    """
    _db[product_id] = {"name": name, "price": price}
    # Invalidate cache for this id to avoid stale data if it existed
    _cache.pop(product_id, None)


def get_product(product_id):
    """Get product by ID, using cache for repeated lookups.

    Returns a shallow copy of the cached/product data or None if not found.
    Returning a copy prevents callers from mutating the internal cache.
    """
    if product_id in _cache:
        return dict(_cache[product_id])
    product = _db.get(product_id)
    if product is not None:
        # store a snapshot in the cache and return a copy
        _cache[product_id] = dict(product)
        return dict(_cache[product_id])
    return None


def update_product(product_id, **fields):
    """Update product fields in the database.

    Invariant: get_product() must return current data after update.
    This function updates the DB and refreshes the cache entry so the
    invariant holds.
    """
    if product_id not in _db:
        return False
    _db[product_id].update(fields)
    # Refresh the cached snapshot so get_product returns current data
    _cache[product_id] = dict(_db[product_id])
    return True


def reset():
    """Clear all data."""
    _db.clear()
    _cache.clear()

```
