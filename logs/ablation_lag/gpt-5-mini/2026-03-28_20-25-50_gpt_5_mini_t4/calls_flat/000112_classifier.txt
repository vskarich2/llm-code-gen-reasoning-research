"""Metrics pipeline: compute raw stats before transforming data."""


def compute_raw_stats(data):
    """Compute statistics on raw (untransformed) data."""
    if not data:
        return {"raw_max": 0, "raw_min": 0, "raw_sum": 0}
    return {
        "raw_max": max(data),
        "raw_min": min(data),
        "raw_sum": sum(data),
    }


def normalize(data):
    """Normalize data to 0-1 range."""
    if not data:
        return []
    lo, hi = min(data), max(data)
    if hi == lo:
        return [0.5] * len(data)
    return [(x - lo) / (hi - lo) for x in data]


def pipeline(data):
    """Run pipeline: raw stats on original data, then normalize.

    raw_stats MUST reflect the original data, not the normalized version.
    """
    # Materialize input to a list so both raw stats and normalization see the
    # same values (avoids iterator exhaustion if an iterator/generator was
    # provided). Note: this may increase memory usage for very large inputs.
    data_list = list(data) if data is not None else []

    raw_stats = compute_raw_stats(data_list)
    normalized = normalize(data_list)
    return {"raw_stats": raw_stats, "cleaned": normalized}


def format_report(result):
    """Format pipeline result for display."""
    return f"max={result['raw_stats']['raw_max']}"
