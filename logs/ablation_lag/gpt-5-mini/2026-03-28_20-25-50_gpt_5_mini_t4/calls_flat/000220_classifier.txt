def run_steps(steps):

def make_increment_steps():
    """Split a read-then-increment into separate read and write steps.

    BUG fixed: step_write uses the current global value at write time

    that each write increments whatever the latest value is, avoiding
    lost updates under any interleaving of read and write steps.
