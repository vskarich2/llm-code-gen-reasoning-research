"""Feature flag system with snake_case keys."""

FLAGS = {
    "dark_mode": True,
    "beta_features": False,
    "new_dashboard": True,
    "analytics_v2": False,
}


def camel_to_snake(name):
    """Convert camelCase string to snake_case."""
    import re
    s1 = re.sub('(.)([A-Z][a-z]+)', r'\\1_\\2', name)
    return re.sub('([a-z0-9])([A-Z])', r'\\1_\\2', s1).lower()


def is_enabled(flag_name):
    """Check if a feature flag is enabled.

    Args:
        flag_name: the flag key to look up (can be camelCase or snake_case)

    Returns:
        True if the flag is enabled, False otherwise.
    """
    normalized_name = camel_to_snake(flag_name)
    return FLAGS.get(normalized_name, False)


def list_flags():
    """Return all flag names."""
    return list(FLAGS.keys())
