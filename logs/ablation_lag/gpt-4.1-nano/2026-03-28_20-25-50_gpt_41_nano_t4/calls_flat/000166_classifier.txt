FLAGS = {
    "dark_mode": True,
    "beta_features": False,
    "new_dashboard": True,
    "analytics_v2": False,
}


def is_enabled(flag_name):
    """Check if a feature flag is enabled.

    Args:
        flag_name: the flag key to look up (must be snake_case)

    Returns:
        True if the flag is enabled, False otherwise.
    """
    # Convert flag_name to snake_case to match FLAGS keys
    snake_case_name = _to_snake_case(flag_name)
    return FLAGS.get(snake_case_name, False)


def list_flags():
    """Return all flag names."""
    return list(FLAGS.keys())


def _to_snake_case(name):
    """Convert camelCase or other formats to snake_case."""
    import re
    s1 = re.sub('(.)([A-Z][a-z]+)', r'\\1_\\2', name)
    snake_case = re.sub('([a-z0-9])([A-Z])', r'\\1_\\2', s1).lower()
    return snake_case
