def with_history(func):
    """Decorator that records call history for a function."""
    def wrapper(*args, **kwargs):
        if not hasattr(wrapper, 'history'):
            wrapper.history = []
        result = func(*args, **kwargs)
        wrapper.history.append({"func": func.__name__, "args_count": len(args)})
        return result
    wrapper.get_history = lambda: list(getattr(wrapper, 'history', []))
    wrapper.clear_history = lambda: getattr(wrapper, 'history', []).clear()
    return wrapper

@with_history
