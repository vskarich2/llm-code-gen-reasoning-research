    "dark_mode": True,
    "beta_features": False,
    "new_dashboard": True,
    "analytics_v2": False,
}

def camel_to_snake(name):
    """Convert camelCase to snake_case."""
    import re
    return re.sub(r'(?<!^)(?=[A-Z])', '_', name).lower()


def is_enabled(flag_name):
    """Check if a feature flag is enabled.

    Args:
        flag_name: the flag key to look up (can be camelCase or snake_case)

    Returns:
        True if the flag is enabled, False otherwise.
    """
    snake_case_flag = camel_to_snake(flag_name)
    return FLAGS.get(snake_case_flag, False)


def list_flags():
    """Return all flag names."""
    return list(FLAGS.keys())
