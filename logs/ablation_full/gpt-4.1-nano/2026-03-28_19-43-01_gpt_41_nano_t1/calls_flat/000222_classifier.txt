_accounts = {}


def reset():
    _accounts.clear()


def create_account(name, balance):
    _accounts[name] = {'balance': balance, 'pending': False}


def get_balance(name):
    account = _accounts.get(name)
    if account:
        return account['balance']
    return 0


def check_balance(name, amount):
    account = _accounts.get(name)
    if account:
        # Check if account is not pending withdrawal and has sufficient balance
        return (not account['pending']) and (account['balance'] >= amount)
    return False


def do_withdraw(name, amount):
    account = _accounts.get(name)
    if account:
        # Deduct amount and clear pending flag
        account['balance'] -= amount
        account['pending'] = False


def run_steps(steps):
    results = []
    for fn, args in steps:
        results.append(fn(*args))
    return results


def make_withdraw_steps(name, amount):
    """Split withdrawal into separate check and act steps, with reservation."""
    result = {"approved": False}

    def step_check():
        account = _accounts.get(name)
        if account and not account['pending'] and account['balance'] >= amount:
            account['pending'] = True  # Reserve the amount
            result["approved"] = True
        else:
            result["approved"] = False
        return ("check", result["approved"])

    def step_act():
        if result["approved"]:
            do_withdraw(name, amount)
        else:
            # If not approved, ensure pending is cleared
            account = _accounts.get(name)
            if account:
                account['pending'] = False
        return ("act", result["approved"])

    return step_check, step_act
