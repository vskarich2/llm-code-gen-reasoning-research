## Failure Mechanism

"""Tabular report with headers and rows."""

from data import make_row, validate_row


class Report:
    """A simple table with column headers and data rows."""

    def __init__(self, headers):
        self.headers = list(headers)
        self.rows = []

    def add_row(self, *values):
        """Add a row of data.

        Enforce that the new row has the same number of columns as headers
        to keep the internal invariant len(row) == len(self.headers).
        """
        row = make_row(*values)
        if not validate_row(row, len(self.headers)):
            raise ValueError(f"Row has {len(row)} columns but expected {len(self.headers)}")
        self.rows.append(row)

    def delete_column(self, index):
        """Remove a column by index from headers and all rows.

        Must remove the header AND the corresponding element
        from every row to keep them aligned.
        """
        # Remove the header at index (may raise IndexError if invalid)
        del self.headers[index]
        # Remove the corresponding element from each row. If a row is already
        # malformed (shorter than expected), raise an informative IndexError so
        # the caller can detect the pre-existing corruption.
        for i, row in enumerate(self.rows):
            try:
                del row[index]
            except IndexError:
                raise IndexError(
                    f"Row {i} has {len(row)} columns; cannot delete column index {index}"
                ) from None

    def render(self):
        """Render the table as a list of dicts keyed by header."""
        result = []
        for row in self.rows:
            result.append(dict(zip(self.headers, row)))
        return result

    def get_column(self, index):
        """Return all values in a given column."""
        return [row[index] for row in self.rows]

    def column_count(self):
        """Return the number of columns based on headers."""
        return len(self.headers)
