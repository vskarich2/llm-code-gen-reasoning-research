    """Return a copy of recorded snapshots."""

def process_batch(items):
    """Process each item: increment counter and snapshot after each.

    For each item in items we:
    - add the item to the running _counter
    - increment _processed_count
    - record a snapshot of the counter state

    Returns the updated _counter.
    """

    for item in items:
        _counter += item
        _processed_count += 1
