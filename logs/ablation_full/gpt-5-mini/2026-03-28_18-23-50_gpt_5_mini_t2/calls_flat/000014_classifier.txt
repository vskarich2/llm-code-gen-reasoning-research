def update_profile(user, changes):
    """Update user profile fields from a changes dict.

    Invariant: display_name must always equal name.

    Behavior:
    - If changes contains 'name' or 'display_name', both user['name'] and
      user['display_name'] are set to the same value (prefer 'name' if both
      are present).
    - Updates 'email' and 'age' when present.
    - Ignores unknown keys.

    The function mutates and returns the provided user mapping.
    """
    # If caller provided a name (or a display_name), compute the canonical
    # name value and apply it to both fields to preserve the invariant.
    if 'name' in changes or 'display_name' in changes:
        name_value = changes.get('name', changes.get('display_name'))
        user['name'] = name_value
        user['display_name'] = name_value

    # Update other supported fields.
    for key, value in changes.items():
        if key in ('name', 'display_name'):
            # already handled
            continue
        if key == 'email':
            user['email'] = value
        elif key == 'age':
            user['age'] = value
        else:
            # Unknown keys are ignored to avoid accidental mutation.
            continue

    return user
