```python
﻿"""Product catalog with caching.

This module stores products in a simple in-memory _db and keeps a _cache
for repeated lookups. The cache is kept consistent on add/update/reset.
get_product returns a shallow copy so callers cannot mutate cached state.
"""

_db = {}
_cache = {}


def add_product(product_id, name, price):
    """Add or overwrite a product in the database and update the cache.

    Ensures that a subsequent get_product(product_id) returns the current
    data (cache consistent with _db).
    """
    _db[product_id] = {"name": name, "price": price}
    # Keep cache consistent with the DB for immediate reads
    _cache[product_id] = dict(_db[product_id])


def get_product(product_id):
    """Get product by ID, using cache for repeated lookups.

    Always returns a shallow copy of the stored mapping to avoid allowing
    callers to mutate the cached object.
    Returns None if the product does not exist in the database.
    """
    if product_id in _cache:
        # Return a copy so callers cannot mutate the cached dict
        return dict(_cache[product_id])
    product = _db.get(product_id)
    if product is not None:
        # Cache a copy of the DB row for future lookups
        _cache[product_id] = dict(product)
        return dict(_cache[product_id])
    return None


def update_product(product_id, **fields):
    """Update product fields in the database and refresh the cache.

    Invariant: get_product() must return current data after update.
    Returns True when the product existed and was updated, False otherwise.
    """
    if product_id not in _db:
        return False
    # Update the DB row (in-place), then refresh the cache to reflect the
    # new DB state. Replacing the cached dict prevents stale reads.
    _db[product_id].update(fields)
    _cache[product_id] = dict(_db[product_id])
    return True


def reset():
    """Clear all data and cache."""
    _db.clear()
    _cache.clear()

```
