 """Authorization handler: decides what a role can do."""

from middleware import authenticate
from roles import get_role_level


def authorize(request):
    """Authorize a request by checking middleware + handler logic.

    Returns dict with 'can_read', 'can_write', 'can_admin'.
    """
    ctx = authenticate(request)
    if not ctx.get("allowed"):
        return {"can_read": False, "can_write": False, "can_admin": False}

    role = ctx["role"]
    level = get_role_level(role)

    # Canonicalized policy derived from numeric role level with explicit
    # special-cases preserved for existing roles (moderator has admin but no write).
    # - can_read: any allowed role
    # - can_write: roles with level >= 40 except moderator
    # - can_admin: roles with level >= 60
    can_read = True
    can_write = False
    can_admin = False

    if level >= 60:
        can_admin = True

    if level >= 40 and role != "moderator":
        can_write = True

    return {"can_read": can_read, "can_write": can_write, "can_admin": can_admin}


def get_authorization_summary(request):
    """Return a human-readable summary of authorization."""
    # Use middleware-resolved role for an accurate summary (avoid request vs ctx mismatch).
    ctx = authenticate(request)
    result = authorize(request)
    perms = [k for k, v in result.items() if v]
    return {"role": ctx.get("role"), "permissions": perms}
