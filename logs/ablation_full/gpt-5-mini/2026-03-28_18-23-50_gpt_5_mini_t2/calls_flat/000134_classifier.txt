'''Simple rate limiter.'''


def is_rate_limited(count, limit):
    """Return True if the caller should be blocked.

    Args:
        count (int): current number of requests already made (>= 0)
        limit (int): maximum allowed requests (must be > 0)

    Returns:
        bool: True if the request should be blocked (count >= limit), False otherwise.

    Raises:
        TypeError: if count or limit is not an int.
        ValueError: if count < 0 or limit <= 0.
    """
    if not isinstance(count, int) or not isinstance(limit, int):
        raise TypeError("count and limit must be integers")
    if count < 0:
        raise ValueError("count must be >= 0")
    if limit <= 0:
        raise ValueError("limit must be > 0")

    # Block when the number of requests already made is greater than or equal
    # to the configured limit. Using '>=' fixes the previous off-by-one bug
    # where count == limit was incorrectly allowed.
    return count >= limit


def check_and_increment(current_count, limit):
    """Check rate limit and return (blocked, new_count).

    If not blocked, increments the count and returns the incremented value.
    """
    if is_rate_limited(current_count, limit):
        return True, current_count
    return False, current_count + 1
