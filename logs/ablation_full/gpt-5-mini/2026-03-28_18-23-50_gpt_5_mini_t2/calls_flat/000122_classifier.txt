```python
"""Access control: map roles to permission sets.

This module provides a clear, immutable mapping from canonical role names
(lowercase strings) to permission sets. It normalizes input role strings,
accepts a small set of aliases, and returns immutable permission sets so
callers cannot accidentally mutate the canonical permissions.
"""

# Immutable empty set used as a stable default (avoids callers mutating a returned set)
FROZEN_EMPTY = frozenset()

# Canonical role -> permission frozenset (all keys are lowercase)
ROLE_PERMISSIONS = {
    "admin": frozenset({"read", "write", "delete", "manage_users"}),
    "moderator": frozenset({"read", "write", "delete"}),
    "user": frozenset({"read", "write"}),
}

# Common aliases resolved to canonical role names (all keys lowercase)
ROLE_ALIASES = {
    "mod": "moderator",
}


def get_permissions(role):
    """Return an immutable set of permissions for the given role.

    Behaviors:
    - Role lookup is case-insensitive.
    - Recognized aliases (e.g. 'mod') are resolved to canonical roles.
    - Non-string or unknown roles return an immutable empty frozenset.
    """
    if not isinstance(role, str):
        return FROZEN_EMPTY

    key = role.lower()
    key = ROLE_ALIASES.get(key, key)
    return ROLE_PERMISSIONS.get(key, FROZEN_EMPTY)


def has_permission(role, action):
    """Check whether a role has a specific permission.

    Returns True iff the permission 'action' is a member of the permission
    set returned by get_permissions(role).
    """
    return action in get_permissions(role)

```
