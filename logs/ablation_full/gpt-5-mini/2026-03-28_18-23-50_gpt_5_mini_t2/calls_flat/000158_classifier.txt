"""Report builder using a single entries list for clarity and correctness.

This refactor replaces the previous parallel lists (_labels and _values)
with a single _entries list holding (label, value) tuples. That prevents
misalignment bugs that could occur if labels and values were mutated
separately.
"""

_entries = []


def add_entry(label, value, position=None):
    """Add an entry to the report.

    If position is given, insert at that index. Otherwise, append to the end.
    Position must be an int or None.
    """
    if position is None:
        _entries.append((label, value))
    else:
        if not isinstance(position, int):
            raise TypeError("position must be an int or None")
        _entries.insert(position, (label, value))


def get_entry(index):
    """Return (label, value) at the given index."""
    return _entries[index]


def get_all():
    """Return list of (label, value) tuples."""
    return list(_entries)


def count():
    """Return the number of entries."""
    return len(_entries)


def reset():
    """Clear all entries."""
    _entries.clear()
