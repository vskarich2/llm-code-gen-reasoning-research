def get_host():
    """Return the current host setting."""
    # Read the current value from the settings dict each call to avoid
    # returning a stale, import-time captured value.
    return _settings["host"]


def get_settings():
    """Return the current settings dict as a shallow copy."""
    return dict(_settings)


def reset_settings():
    """Reset settings to default values in-place.

    Mutating the existing dict preserves identity for any external code that
    may hold a reference to _settings.
    """
    # Mutate the existing dict instead of rebinding the global name.
    _settings.clear()
    _settings.update({"host": "localhost", "port": 8080, "debug": False})


def configure(host=None, port=None, debug=None):
    """Update settings.

    Only updates keys when the corresponding argument is not None. This
    preserves existing values when callers pass None explicitly.
    """
    if host is not None:
        _settings["host"] = host
    if port is not None:
        _settings["port"] = port
    if debug is not None:
