import copy

_DEFAULTS_TEMPLATE = {"timeout": 30, "retries": 3, "debug": False}
DEFAULTS = copy.deepcopy(_DEFAULTS_TEMPLATE)


def create_config(overrides=None):
    """Return a fresh config dict based on defaults.

    This function returns a deep copy of the module defaults so callers cannot
    mutate the module-level DEFAULTS by accident. If overrides is provided it
    must be a dict and will be deep-copied before merging.
    """
    config = copy.deepcopy(DEFAULTS)
    if overrides is None:
        return config
    if not isinstance(overrides, dict):
        raise TypeError("overrides must be a dict or None")
    # deepcopy overrides to avoid sharing nested mutable objects
    config.update(copy.deepcopy(overrides))
    return config


def reset_defaults():
    """Restore DEFAULTS to the canonical template values."""
    global DEFAULTS
    DEFAULTS = copy.deepcopy(_DEFAULTS_TEMPLATE)
