```python
import re
import warnings

"""Feature flag system with snake_case keys.

This module stores flags in snake_case. The public helpers accept either
snake_case or camelCase flag names; camelCase names are normalized to
snake_case before lookup to avoid silent lookup failures.
"""

FLAGS = {
    "dark_mode": True,
    "beta_features": False,
    "new_dashboard": True,
    "analytics_v2": False,
}


def _to_snake_case(name):
    """Normalize a flag name to snake_case.

    Rules implemented:
    - If name is not a string, raise TypeError (avoid silent behavior).
    - If name already contains underscores and is lowercase, return as-is.
    - Convert camelCase (e.g. 'darkMode') and mixedCase to snake_case
      (e.g. 'dark_mode'). Hyphens are treated as underscores.

    This is a conservative, deterministic transformation; unusual
    acronyms may produce underscored letters (e.g. 'HTMLParser' ->
    'h_t_m_l_parser'). Callers will receive a warning if the normalized
    name does not exist in FLAGS.
    """
    if not isinstance(name, str):
        raise TypeError("flag_name must be a string")

    # Fast path: already snake_case-like
    if '_' in name and name == name.lower():
        return name

    # Replace hyphens with underscores to avoid losing separators
    s = name.replace('-', '_')

    # Insert underscores before transitions from lower/number to Upper
    s = re.sub(r'(.)([A-Z][a-z]+)', r'\1_\2', s)
    s = re.sub(r'([a-z0-9])([A-Z])', r'\1_\2', s)

    return s.lower()


def is_enabled(flag_name):
    """Check whether a feature flag is enabled.

    Accepts either snake_case or camelCase flag_name. This function
    normalizes camelCase inputs to snake_case before lookup.

    Args:
        flag_name: the flag key to look up (snake_case or camelCase)

    Returns:
        True if the flag is enabled, False otherwise.
    """
    normalized = _to_snake_case(flag_name)

    if normalized not in FLAGS:
        # Warn instead of silently returning the default so callers can
        # discover naming mismatches during development.
        warnings.warn(
            "Flag name '%s' normalized to '%s', but no such flag exists in FLAGS." % (flag_name, normalized),
            UserWarning,
        )
        return False

    return bool(FLAGS.get(normalized, False))


def list_flags():
    """Return all flag names (snake_case) in stable order."""
    return sorted(FLAGS.keys())

```
