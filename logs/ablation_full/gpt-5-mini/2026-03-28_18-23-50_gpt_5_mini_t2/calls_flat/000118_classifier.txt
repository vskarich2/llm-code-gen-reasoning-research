Stage 2: normalize
Stage 3: clip to range
Stage 4: summarize for display

from metrics import compute_raw_stats, compute_derived


def pipeline(data):
    """4-stage pipeline.

    raw_stats MUST reflect the original data before any transforms.
    """
    # Materialize the input so that raw stats are computed on the exact
    # original values and so subsequent transforms can iterate the same
    # values even if the caller passed a generator/iterator.
    original = list(data) if not isinstance(data, list) else data[:]

    # Stage 1: raw stats (must run before any transforms)
    raw_stats = compute_raw_stats(original)
    derived = compute_derived(raw_stats)

    # Stage 2: normalize
    normalized = normalize(original)
    # Stage 3: clip
    clipped = clip(normalized, 0.05, 0.95)
    # Stage 4: display summary
    display = summarize_for_display(clipped)

    return {
        "raw_stats": raw_stats,
        "derived": derived,
        "cleaned": clipped,
        "display": display,
    }
