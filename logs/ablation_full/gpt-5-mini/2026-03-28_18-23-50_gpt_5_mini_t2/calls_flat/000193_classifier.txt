from models import Account
from ledger import record_debit, record_credit, record_transfer_attempt
from audit import emit_transfer_event, emit_failure_alert


def validate_transfer(sender, amount):
    if amount <= 0:
        raise ValueError("amount must be positive")
    if sender.balance < amount:
        raise ValueError("insufficient funds")


def _do_transfer(sender, receiver, amount):
    """Core business logic: validate and update both account balances.

    This function keeps a snapshot of the original balances and will
    restore them if an exception occurs during the assignment, so
    callers can rely on it not leaving partial mutations.
    """
    validate_transfer(sender, amount)
    old_sender = sender.balance
    old_receiver = receiver.balance
    try:
        # perform updates; if something goes wrong while assigning, we
        # catch and rollback below
        sender.balance = old_sender - amount
        receiver.balance = old_receiver + amount
    except Exception:
        # rollback to preserve invariant
        try:
            sender.balance = old_sender
        finally:
            receiver.balance = old_receiver
        raise


def execute_transfer(sender, receiver, amount):
    # Keep validation at the top so we never record attempts for invalid data
    validate_transfer(sender, amount)

    # Observability: record an initial pending attempt
    record_transfer_attempt(sender.account_id, receiver.account_id, amount, status="pending")

    try:
        # Simulate a transient failure before committing business state.
        # Doing this here (outside _do_transfer) ensures we don't mutate balances
        # and don't record partial ledger entries when the transient failure occurs.
        if random.random() < 0.3:
            # Emit failure and raise; wrapper catch will mark attempt failed
            emit_failure_alert(
                sender.account_id, receiver.account_id, amount,
                "connection reset during credit phase",
            )
            raise RuntimeError("transient failure during credit")

        # Perform the atomic business mutation
        _do_transfer(sender, receiver, amount)

    except Exception as exc:
        # Observability on failure: mark failed and emit alert (if not already emitted)
        record_transfer_attempt(sender.account_id, receiver.account_id, amount, status="failed")

        # (like the transient failure branch above) will have already emitted it.
        emit_failure_alert(sender.account_id, receiver.account_id, amount, exc)
        raise

        record_debit(sender.account_id, amount)
        record_credit(receiver.account_id, amount)
        record_transfer_attempt(sender.account_id, receiver.account_id, amount, status="completed")
        emit_transfer_event(sender.account_id, receiver.account_id, amount)
