def retry_send(msg, max_retries=2):
    """Send with retry.

    Attempts to send up to max_retries times. Stops and returns True as soon
    as a send(msg) call returns True. If send returns False or raises an
    Exception, the call is treated as a failed attempt and the next attempt
    proceeds. If all attempts fail, returns False.

    Note: max_retries is the total number of attempts; values <= 0 result
    in zero attempts and False is returned.
    """
    for _ in range(max_retries):
        try:
            result = send(msg)
        except Exception:
            # Treat exceptions as a failed attempt; continue to retry.
            result = False

        if result:
            # Success: stop retrying immediately to avoid duplicate sends.
            return True
        # else: result is False -> continue to next attempt

    return False

    return list(_sent)
