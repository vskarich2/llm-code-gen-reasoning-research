import copy

DEFAULTS = {"timeout": 30, "retries": 3, "debug": False}


def create_config(overrides=None):
    """Return a config dict based on defaults, with optional overrides.

    A deep copy of the module-level DEFAULTS is returned so callers cannot
    accidentally mutate the module-level defaults by updating the returned
    dict. If overrides is provided it must be a dict and will be applied on
    top of the copied defaults.
    """
    # Make an independent copy so updates do not mutate the module-level DEFAULTS
    config = copy.deepcopy(DEFAULTS)
    if overrides:
        if not isinstance(overrides, dict):
            raise TypeError("overrides must be a dict")
        config.update(overrides)
    return config


def reset_defaults():
    """Restore defaults to factory values."""
    global DEFAULTS
    DEFAULTS = {"timeout": 30, "retries": 3, "debug": False}
