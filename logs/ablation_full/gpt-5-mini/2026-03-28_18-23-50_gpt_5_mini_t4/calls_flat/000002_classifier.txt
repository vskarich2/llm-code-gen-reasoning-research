DEFAULTS: Dict[str, Any] = _INITIAL_DEFAULTS.copy()


def create_config(overrides: Optional[Dict[str, Any]] = None) -> Dict[str, Any]:
    """Return a new config dict based on DEFAULTS merged with overrides.

    This function never mutates the module-level DEFAULTS dict; it returns a
    fresh dict so callers cannot accidentally change the global defaults.
    """
    config = DEFAULTS.copy()
    if overrides:
        config.update(overrides)
    return config


def reset_defaults() -> None:
    """Restore DEFAULTS to its original values in-place.

    rebinding the name. Mutating in-place preserves identity for any external
    references that may hold the DEFAULTS object.
