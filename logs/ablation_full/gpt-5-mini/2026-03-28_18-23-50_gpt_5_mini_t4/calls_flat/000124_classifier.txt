}


def _normalize_role(role):
    """Normalize role input to a canonical form used as dict keys.

    - Accepts only string roles. Non-strings are treated as unknown.
    - Strips surrounding whitespace and lowercases to handle common caller variations.
    """
    if not isinstance(role, str):
        return None
    return role.strip().lower()


def get_permissions(role):
    """Return an immutable set (frozenset) of permissions for the given role.

    - Normalizes the role input to reduce lookup failures caused by case/whitespace.
    - Returns an empty frozenset for unknown roles.
    - Returning a frozenset prevents callers from mutating module-level permission sets.
    """
    norm = _normalize_role(role)
    if not norm:
        return frozenset()
    perms = ROLE_PERMISSIONS.get(norm)
    if perms:
        return frozenset(perms)
    return frozenset()


def has_permission(role, action):
    """Check whether a role has a specific permission (action).

    Membership checks work with the frozenset returned by get_permissions().
    """
    return action in get_permissions(role)
