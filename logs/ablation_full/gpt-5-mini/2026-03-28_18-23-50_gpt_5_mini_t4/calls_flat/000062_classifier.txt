```python
"""Batch processor with per-item snapshots.

This module maintains a running counter and records a snapshot of the counter
after each processed item. It exposes utilities for resetting state and
verifying that exactly one snapshot was recorded per processed item.
"""

_counter = 0
_snapshots = []
_processed_items = 0


def reset():
    """Reset all module state to initial values."""
    global _counter, _snapshots, _processed_items
    _counter = 0
    _snapshots = []
    _processed_items = 0


def snapshot():
    """Record the current counter value as a snapshot.

    This appends the current value of _counter to the _snapshots list.
    """
    # appending to a list mutates it in-place; no global declaration needed
    _snapshots.append(_counter)


def get_snapshots():
    """Return a shallow copy of the snapshots list."""
    return list(_snapshots)


def process_batch(items):
    """Process each item: increment counter and snapshot after each.

    For each element in items this function increments the running
    _counter by that element's value, records a snapshot immediately
    after that increment, and updates the count of processed items.
    Returns the new counter value.
    """
    global _counter, _processed_items
    for item in items:
        _counter += item
        # record a snapshot for this item
        snapshot()
        # record that we processed one more item
        _processed_items += 1
    return _counter


def verify_consistency():
    """Verify that there is exactly one snapshot per processed item.

    Returns True when len(_snapshots) equals the total number of items
    processed by all prior calls to process_batch (since the last reset()).
    """
    return _processed_items == len(_snapshots)

```
