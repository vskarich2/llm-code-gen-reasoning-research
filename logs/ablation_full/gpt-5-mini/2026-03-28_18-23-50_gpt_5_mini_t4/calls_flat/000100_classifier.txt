 """Order fulfillment with inventory and wallet."""


class Inventory:
    def __init__(self, stock):
        self.stock = stock
        self.reserved = 0

    def reserve(self, qty):
        if qty > self.stock - self.reserved:
            raise ValueError("insufficient stock")
        self.reserved += qty

    def release(self, qty):
        # Prevent releasing more than reserved to avoid underflow/corruption
        if qty > self.reserved:
            raise ValueError("release exceeds reserved")
        self.reserved -= qty

    def finalize(self, qty):
        # Move reserved items into sold (deduct from stock and reserved).
        if qty > self.reserved:
            raise ValueError("finalize exceeds reserved")
        if qty > self.stock:
            # Defensive: should not happen if reserve succeeded, but check anyway
            raise ValueError("finalize exceeds stock")
        self.stock -= qty
        self.reserved -= qty

    def available(self):
        return self.stock - self.reserved


class Wallet:
    def __init__(self, balance):
        self.balance = balance

    def charge(self, amount):
        if amount > self.balance:
            raise ValueError("insufficient funds")
        self.balance -= amount


def place_order(inventory, wallet, qty, price):
    """Reserve inventory, then charge wallet. Release on failure and finalize on success.

    Steps:
    - Reserve the requested qty in inventory (raises on insufficient stock)
    - Attempt to charge the wallet
      - On any exception during charging, release the reservation and re-raise
    - On successful charge, finalize the inventory (decrement stock and clear reservation)
    """
    inventory.reserve(qty)
    try:
        wallet.charge(qty * price)
    except BaseException:
        # Ensure we release the reservation on any failure (including interrupts);
        # re-raise so the caller sees the original error.
        inventory.release(qty)
        raise
    # Charge succeeded: commit the reservation into the stock (finalize)
    inventory.finalize(qty)
    return {"status": "confirmed", "qty": qty, "total": qty * price}
