"""Product catalog with caching."""

import copy

_db = {}
_cache = {}


def add_product(product_id, name, price):
    """Add a product to the database.

    Also update the cache for this product_id to keep the cache consistent
    in case add_product is used to overwrite an existing entry.
    """
    _db[product_id] = {"name": name, "price": price}
    # Keep cache consistent with the database (store a deep copy)
    _cache[product_id] = copy.deepcopy(_db[product_id])


def get_product(product_id):
    """Get product by ID, using cache for repeated lookups.

    Always return a deep copy of cached data to avoid callers mutating
    the internal cache state.
    """
    if product_id in _cache:
        return copy.deepcopy(_cache[product_id])
    product = _db.get(product_id)
    if product is not None:
        # store a deep copy in cache to avoid aliasing nested mutables
        _cache[product_id] = copy.deepcopy(product)
        return copy.deepcopy(_cache[product_id])
    return None


def update_product(product_id, **fields):
    """Update product fields in the database.

    Invariant: get_product() must return current data after update.
    """
    if product_id not in _db:
        return False
    _db[product_id].update(fields)
    # Ensure cache is consistent with the database by overwriting the cached
    # snapshot with a deep copy of the updated DB entry.
    _cache[product_id] = copy.deepcopy(_db[product_id])
    return True


def reset():
    """Clear all data."""
    _db.clear()
    _cache.clear()
