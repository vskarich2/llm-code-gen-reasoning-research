from flags import is_enabled


def compute_price(base, qty, use_new_pricing=None):
    """
    Compute price using either v1 or v2 pricing.
    If use_new_pricing is None, consult the global feature flag; otherwise use the explicit override.
    """
    if use_new_pricing is None:
        enabled = is_enabled("new_pricing")
    else:
        enabled = bool(use_new_pricing)

    if enabled:
        return _v2_price(base, qty)
    return _v1_price(base, qty)


def _v1_price(base, qty):
    return base * qty


def _v2_price(base, qty):
    discount = 0.1 if qty >= 10 else 0
    return base * qty * (1 - discount)


def get_pricing_version(use_new_pricing=None):
    """
    Return 'v2' or 'v1'. Accepts an optional override so callers can force a pricing version without mutating global flags.
    """
    if use_new_pricing is None:
        enabled = is_enabled("new_pricing")
    else:
        enabled = bool(use_new_pricing)
    return "v2" if enabled else "v1"


from pricing import compute_price, get_pricing_version
from flags import is_enabled

_invoices = []


def create_invoice(customer, items, use_new_pricing=None):
    """
    Create an invoice. If use_new_pricing is provided (True/False), it overrides the global 'new_pricing' flag
    for pricing decisions for this invoice only.
    """
    total = 0
    lines = []
    for item in items:
        price = compute_price(item["base"], item["qty"], use_new_pricing)
        lines.append({"sku": item["sku"], "amount": price})
        total += price

    invoice = {
        "customer": customer,
        "lines": lines,
        "total": total,
        "pricing_version": get_pricing_version(use_new_pricing),
    }

    if is_enabled("audit_mode"):
        invoice["reasoning_evaluator_audit"] = True
        _invoices.append(invoice)

    return invoice
