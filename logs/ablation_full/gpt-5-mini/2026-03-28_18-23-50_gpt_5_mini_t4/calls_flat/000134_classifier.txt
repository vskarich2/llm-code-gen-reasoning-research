 """Simple rate limiter."""


def is_rate_limited(count: int, limit: int) -> bool:
    """Return True when the current request count has reached the allowed limit.

    Args:
        count: number of requests that have already been processed (non-negative integer).
        limit: maximum allowed requests (non-negative integer).

    Returns:
        True if the caller should be blocked, False if the request is allowed.
    """
    if count < 0 or limit < 0:
        raise ValueError("count and limit must be non-negative")

    # Block when count is equal to or greater than limit (inclusive).
    return count >= limit


def check_and_increment(count: int, limit: int):
    """Check rate limit and return (blocked, new_count).

    If the caller is not blocked, the function returns (False, count + 1).
    If blocked, it returns (True, count) and does not increment.
    """
    blocked = is_rate_limited(count, limit)
    if blocked:
        return True, count
    return False, count + 1
