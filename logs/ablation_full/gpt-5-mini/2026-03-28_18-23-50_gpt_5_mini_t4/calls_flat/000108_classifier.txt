_notifications = []


def reset():

    _notifications = []
    # delegate resets to the inventory and payment modules to avoid touching
    # local variables that don't exist here
    inventory_reset()
    payment.reset()


def place_order(product_id, qty, price):
    """Place order: reserve inventory, log reasoning_evaluator_audit, process payment, notify.

    If payment fails, must release inventory AND remove reasoning_evaluator_audit entry.
    """
    order_id = f"ORD-{product_id}-{qty}"
    # reserve inventory first
    reserve(product_id, qty)
    # add audit entry so there is a record before attempting payment
    add_audit_entry({"order_id": order_id, "product": product_id, "qty": qty})
    try:
        result = process(qty * price, order_id)
    except Exception as e:
        # Rollback compensating actions robustly: try to release inventory and
        # remove the audit entry, but preserve and re-raise the original exception.
        try:
            # only release if there is at least the reserved quantity recorded
            if get_reserved(product_id) >= qty:
                release(product_id, qty)
        except Exception:
            # swallow secondary errors from rollback to ensure original error
            # is propagated and does not get masked by rollback failures
            pass
        try:
            remove_audit_entry(order_id)
        except Exception:
            pass
        raise
    _notifications.append({"order_id": order_id, "status": "confirmed"})
    return {"status": "confirmed", "payment": result}
