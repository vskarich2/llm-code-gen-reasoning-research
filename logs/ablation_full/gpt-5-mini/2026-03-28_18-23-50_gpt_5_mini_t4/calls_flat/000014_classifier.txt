"""User profile management."""


def update_profile(user, changes):
    """Update user profile fields from changes dict.

    Invariant: display_name must always equal name. Any change to 'name' or
    'display_name' will update both fields so they remain identical.
    Unknown keys are ignored.
    """
    for key, value in changes.items():
        if key == "name" or key == "display_name":
            # Keep both fields in sync so the invariant holds
            user["name"] = value
            user["display_name"] = value
        elif key == "email":
            user["email"] = value
        elif key == "age":
            user["age"] = value
    return user

        "display_name": name,
        "email": email,

    }
