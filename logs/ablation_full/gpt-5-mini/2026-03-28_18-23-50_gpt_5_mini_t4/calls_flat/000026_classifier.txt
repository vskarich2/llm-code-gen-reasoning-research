```python
'''Product catalog with caching.

This module maintains an in-memory product database (_db) and a simple read cache
(_cache). Functions ensure the cache stays consistent with the database: when a
product is added or updated we update the cache; get_product always returns a
fresh shallow copy so callers cannot mutate the cached object.
'''

_db = {}
_cache = {}


def add_product(product_id, name, price):
    """Add or replace a product in the database and update the cache.

    After this call, get_product(product_id) will reflect the newly added
    product.
    """
    _db[product_id] = {"name": name, "price": price}
    # Keep cache consistent with the database for immediate reads
    _cache[product_id] = dict(_db[product_id])


def get_product(product_id):
    """Get product by ID, using cache for repeated lookups.

    Returns a shallow copy of the cached product dict or None if not found.
    Returning a copy prevents callers from mutating the cached object directly.
    """
    if product_id in _cache:
        return dict(_cache[product_id])
    product = _db.get(product_id)
    if product is not None:
        _cache[product_id] = dict(product)
        return dict(_cache[product_id])
    return None


def update_product(product_id, **fields):
    """Update product fields in the database and keep cache consistent.

    Returns True on success, False if the product does not exist.
    Invariant: get_product() must return current data after update.
    """
    if product_id not in _db:
        return False
    _db[product_id].update(fields)
    # Keep cache consistent: update the cached copy if present
    if product_id in _cache:
        _cache[product_id].update(fields)
    return True


def reset():
    """Clear all data."""
    _db.clear()
    _cache.clear()

```
