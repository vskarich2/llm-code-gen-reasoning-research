```python
"""Feature flag system with snake_case keys.

This module normalizes caller-provided flag names (camelCase, PascalCase, or
snake_case) to snake_case before looking them up in FLAGS. Unknown flags return
False (preserving previous behavior) but emit a RuntimeWarning so tests and
logs can surface typos or mismatches.
"""

import re
import warnings

FLAGS = {
    "dark_mode": True,
    "beta_features": False,
    "new_dashboard": True,
    "analytics_v2": False,
}


def _to_snake_case(name):
    """Convert camelCase/PascalCase/hyphenated names to snake_case.

    Rules:
    - Accepts a string and returns a lower-case, underscore-separated name.
    - Replaces '-' with '_'.
    - Converts CamelCase / camelCase to snake_case using common regex rules.

    This helper is intentionally small and deterministic (pure function).
    """
    if not isinstance(name, str):
        raise TypeError(f"flag_name must be str, got {type(name).__name__}")

    s = name.replace("-", "_")

    # Fast path: already all-lower and contains underscore (likely snake_case)
    if s == s.lower() and "_" in s:
        return s

    # Convert CamelCase/PascalCase to snake_case
    # Step 1: FooBar -> Foo_Bar (handles boundary between lowercase and Uppercase groups)
    s1 = re.sub(r"(.)([A-Z][a-z]+)", r"\1_\2", s)
    # Step 2: fooBar -> foo_Bar (handles boundary where a lowercase/digit is followed by Uppercase)
    snake = re.sub(r"([a-z0-9])([A-Z])", r"\1_\2", s1).lower()

    return snake


def is_enabled(flag_name):
    """Check if a feature flag is enabled.

    Accepts the flag name in camelCase, PascalCase, or snake_case.

    Returns True if the normalized flag exists in FLAGS and is truthy, False
    otherwise. If the normalized name is not present in FLAGS a RuntimeWarning
    is emitted to help detect typos or naming mismatches.
    """
    if not isinstance(flag_name, str):
        raise TypeError(f"flag_name must be str, got {type(flag_name).__name__}")

    normalized = _to_snake_case(flag_name)

    if normalized in FLAGS:
        # Coerce to bool to avoid returning non-bool truthy values accidentally
        return bool(FLAGS[normalized])

    # Preserve the previous silent-default behavior but surface a warning
    warnings.warn(
        f"Unknown feature flag '{flag_name}' (normalized to '{normalized}'). Returning False.",
        RuntimeWarning,
    )
    return False


def list_flags():
    """Return all stored flag names (snake_case).

    The returned list reflects the current keys of FLAGS and may change if
    FLAGS is mutated at runtime.
    """
    return list(FLAGS.keys())

```
