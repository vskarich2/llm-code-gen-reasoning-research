"""Feature flag system with snake_case keys."""

import logging
import re

# Keep module-level flags mapping as the source of truth.
FLAGS = {
    "dark_mode": True,
    "beta_features": False,
    "new_dashboard": True,
    "analytics_v2": False,
}


def _normalize_flag_name(flag_name):
    """Normalize a flag name to snake_case.

    - If the input is already snake_case, it will be returned lowercased.
    - If the input is camelCase (e.g. "darkMode"), it will be converted to
      snake_case ("dark_mode").
    - This function is a pure transformation and does not access module state.
    """
    # Basic camelCase -> snake_case conversion using regex. Handles common
    # transitions between lower/upper and number sequences.
    s = flag_name
    # Insert underscores between lower->upper and upper->lower/number boundaries
    s = re.sub(r"(.)([A-Z][a-z]+)", r"\1_\2", s)
    s = re.sub(r"([a-z0-9])([A-Z])", r"\1_\2", s)
    return s.replace('-', '_').lower()


def is_enabled(flag_name):
    """Check if a feature flag is enabled.

    Args:
        flag_name: the flag key to look up (accepts snake_case or camelCase)

    Returns:
        True if the flag is enabled, False otherwise.

    Raises:
        TypeError: if flag_name is not a str.
    """
    if not isinstance(flag_name, str):
        raise TypeError("flag_name must be a string")

    # Normalize camelCase -> snake_case so callers using different naming
    # conventions still resolve to the same canonical keys stored in FLAGS.
    key = _normalize_flag_name(flag_name)

    if key not in FLAGS:
        # Keep the original module behavior of returning False for unknown
        # flags, but add a debug log to make unexpected lookups visible.
        logging.debug("Feature flag lookup for unknown key: %s (normalized from %s)", key, flag_name)
        return False

    return bool(FLAGS[key])


def list_flags():
    """Return all flag names."""
    return list(FLAGS.keys())
