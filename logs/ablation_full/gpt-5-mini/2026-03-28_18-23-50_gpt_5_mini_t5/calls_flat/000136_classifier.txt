def is_rate_limited(count, limit):
    """Return True if the caller should be blocked given the current count and limit.

    Args:
        count (int): current number of requests made (non-negative).
        limit (int): maximum allowed requests (non-negative).

    Returns:
        bool: True if requests should be blocked (count >= limit), False otherwise.

    Raises:
        TypeError: if count or limit are not integers.
        ValueError: if count or limit are negative.
    """
    # Input validation makes behavior explicit and avoids silent off-by-one surprises
    if not isinstance(count, int) or not isinstance(limit, int):
        raise TypeError("count and limit must be integers")
    if count < 0 or limit < 0:
        raise ValueError("count and limit must be non-negative")

    # Correct comparison: block when count >= limit
    return count >= limit


def check_and_increment(current_count, limit):
    """Check the rate limit and return (blocked, new_count).

    If blocked, returns True and the current_count unchanged.

    Args:
        current_count (int): current number of requests made (non-negative).
        limit (int): maximum allowed requests (non-negative).

    Returns:
        (bool, int): (blocked, new_count)
    """
