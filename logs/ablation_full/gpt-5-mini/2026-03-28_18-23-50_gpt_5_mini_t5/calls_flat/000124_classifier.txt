ROLE_PERMISSIONS = {
    "admin": {"read", "write", "delete", "manage_users"},
    "user": {"read", "write"},
    "moderator": {"read", "write", "delete"},
}


def get_permissions(role):
    """Return the set of permissions for the given role.

    Behavior:
    - Normalizes role strings (strip + lower) so common caller variants like
      'Moderator' or ' moderator ' resolve correctly.
    - Returns an immutable frozenset so callers cannot mutate the internal
      ROLE_PERMISSIONS values.
    """
    if not isinstance(role, str):
        return frozenset()
    key = role.strip().lower()
    return frozenset(ROLE_PERMISSIONS.get(key, set()))


def has_permission(role, action):
    """Check whether a role has a specific permission."""
    if not isinstance(action, str):
        return False
    return action in get_permissions(role)
