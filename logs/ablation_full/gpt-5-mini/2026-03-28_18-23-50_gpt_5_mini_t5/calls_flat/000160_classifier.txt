## Fix Strategy

Add a post-condition assertion that len(_labels) == len(_values) to catch future regressions. This breaks the causal chain by ensuring the label and value are mutated in lock-step at the same index and immediately verifying the invariant.

"""Report builder with parallel label/value arrays.

This module maintains two parallel lists: _labels and _values.
Each index i must correspond: _labels[i] matches _values[i].

The previous implementation inserted into _labels but appended to _values when
position was provided, which caused misalignment. add_entry now inserts into
both lists at the same position and validates the invariant.
"""

_labels = []
_values = []


def add_entry(label, value, position=None):
    """Add an entry to the report.

    If position is given, insert both label and value at that index.
    Otherwise, append to the end.

    A post-condition assertion verifies that the two parallel lists remain
    the same length after the operation.
    """
    if position is None:
        _labels.append(label)
        _values.append(value)
    else:
        # Insert label and value at the same index to keep lists aligned.
        _labels.insert(position, label)
        _values.insert(position, value)

    # Invariant check: lengths must match after any mutation.
    assert len(_labels) == len(_values), "Invariant violated: labels/values length mismatch"


def get_entry(index):
    """Return (label, value) at the given index."""
    return (_labels[index], _values[index])

    """Return list of (label, value) tuples."""
    return list(zip(_labels, _values))


def count():

    return len(_labels)


def reset():
    """Clear all entries."""
    _labels.clear()
    _values.clear()
