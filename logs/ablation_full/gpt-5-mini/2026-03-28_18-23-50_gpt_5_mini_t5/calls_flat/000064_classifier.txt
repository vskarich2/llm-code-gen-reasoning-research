    _snapshots = []
    _processed_count = 0


def snapshot():
    """Record the current counter value by appending to _snapshots.

    snapshot() uses the current module-level _counter and appends it to
    the _snapshots list. It does not modify _processed_count; callers
    are responsible for incrementing processed count after a successful
    snapshot to keep invariants consistent.
    """
    _snapshots.append(_counter)


def get_snapshots():
    return list(_snapshots)


def process_batch(items):
    """Process each item: increment counter and snapshot after each.

    For every item in items we perform the sequence:
      1) add item to _counter
      2) record a snapshot of the new _counter value
      3) increment _processed_count to reflect a successfully processed
         item that has a corresponding snapshot

    This ordering ensures that if snapshot() fails, we do not increment
    _processed_count and thus preserve the invariant that
    len(_snapshots) == _processed_count.
    """
    global _counter, _processed_count
    for item in items:
        _counter += item
        # Make a per-item snapshot immediately after the increment.
        snapshot()
        # Only count the item as processed after the snapshot succeeded.
        _processed_count += 1
    return _counter


def verify_consistency():
    """Each processed item should produce one snapshot.

    Returns True when the number of snapshots equals the number of items
    successfully processed (tracked by _processed_count).
    """
    return _processed_count == len(_snapshots)
