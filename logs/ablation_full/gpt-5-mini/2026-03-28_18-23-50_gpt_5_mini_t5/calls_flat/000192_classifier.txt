# Code Produced
```python
from billing import create_invoice, get_invoices
from flags import enable, disable, is_enabled


def checkout(customer, items, use_new_pricing=False):
    # Preserve previous flag state so we don't permanently mutate global flags
    prev = is_enabled("new_pricing")
    try:
        if use_new_pricing:
            enable("new_pricing")
        else:
            disable("new_pricing")
        invoice = create_invoice(customer, items)
    finally:
        # Restore previous state even if create_invoice raises
        if prev:
            enable("new_pricing")
        else:
            disable("new_pricing")

    return invoice


def admin_report():
    return get_invoices()
```
