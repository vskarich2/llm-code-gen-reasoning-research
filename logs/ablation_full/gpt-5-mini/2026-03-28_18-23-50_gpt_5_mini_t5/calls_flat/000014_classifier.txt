from typing import Any, Dict, Optional


def _set_name(user: Dict[str, Any], name: str) -> None:

    This enforces the invariant that display_name always equals name.
    """
    user["name"] = name
    user["display_name"] = name


def update_profile(user: Dict[str, Any], changes: Dict[str, Any]) -> Dict[str, Any]:
    """Update user profile fields from changes dict.

    Invariant: display_name must always equal name. To preserve that,
    callers must update the 'name' field; direct updates to 'display_name'
    are rejected.

    Supported keys in changes: 'name', 'email', 'age'. Unknown keys raise
    KeyError to make unexpected input explicit.
    """
    if "display_name" in changes:
        # display_name is derived from name; do not allow direct edits
        raise ValueError("'display_name' is derived from 'name'; set 'name' instead")

    for key, value in changes.items():
        if key == "name":
            _set_name(user, value)
        elif key == "email":
            user["email"] = value
        elif key == "age":
            user["age"] = value
        else:
            # Reject unknown fields to avoid silent failures
            raise KeyError(f"Unknown profile field: {key}")

    return user


def create_user(name: str, email: str) -> Dict[str, Optional[Any]]:
    """Create a new user profile with synced display_name."""
    return {
        "name": name,
        "display_name": name,
        "email": email,
        "age": None,
    }
