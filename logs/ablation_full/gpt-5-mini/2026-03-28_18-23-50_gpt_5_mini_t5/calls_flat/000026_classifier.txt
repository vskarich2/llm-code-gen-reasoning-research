```python
"""Product catalog with a simple cache.

This module provides a small in-memory product store (_db) with a read cache
(_cache). The cache stores shallow copies of product dictionaries so callers
don't accidentally mutate the internal store. When a product is changed via
update_product, the cache for that product is invalidated so subsequent
get_product calls repopulate the cache from the authoritative _db.
"""

_db = {}
_cache = {}


def add_product(product_id, name, price):
    """Add a product to the database.

    product_id: hashable key for the product
    name: product name (string)
    price: product price (numeric)
    """
    _db[product_id] = {"name": name, "price": price}


def get_product(product_id):
    """Get product by ID, using cache for repeated lookups.

    Returns a dict copy of the product data or None if not found.
    """
    if product_id in _cache:
        return _cache[product_id]
    product = _db.get(product_id)
    if product is not None:
        # Store a shallow copy in the cache so callers don't get a direct
        # reference to the internal _db entry.
        _cache[product_id] = dict(product)
    return _cache.get(product_id)


def update_product(product_id, **fields):
    """Update product fields in the database.

    Invariant: get_product() must return current data after update.
    Returns True if the product existed and was updated, False otherwise.
    """
    if product_id not in _db:
        return False
    _db[product_id].update(fields)
    # Invalidate any cached copy for this product so get_product will
    # fetch and cache the up-to-date data from _db.
    _cache.pop(product_id, None)
    return True


def reset():
    """Clear all data and cache."""
    _db.clear()
    _cache.clear()

```
