 """Order fulfillment with inventory and wallet."""


class Inventory:
    def __init__(self, stock):
        self.stock = stock
        self.reserved = 0

    def reserve(self, qty):
        if qty > self.stock - self.reserved:
            raise ValueError("insufficient stock")
        self.reserved += qty

    def release(self, qty):
        """Release a previous reservation.

        Validate to avoid underflow: releasing more than reserved is a logic error.
        """
        if qty < 0:
            raise ValueError("cannot release negative quantity")
        if qty > self.reserved:
            raise ValueError("release exceeds reserved quantity")
        self.reserved -= qty

    def available(self):
        return self.stock - self.reserved


class Wallet:
    def __init__(self, balance):
        self.balance = balance

    def charge(self, amount):
        if amount > self.balance:
            raise ValueError("insufficient funds")
        self.balance -= amount


def place_order(inventory, wallet, qty, price):
    """Reserve inventory, then charge wallet. Release on failure.

    Behavior:
    - If inventory.reserve(qty) raises, no payment attempt is made.
    - If wallet.charge(...) raises, the reservation is released and the
      original exception is re-raised.
    - On success, reservation remains (order confirmed) — caller is
      responsible for later fulfillment or explicit release if desired.
    """
    # Acquire reservation first; if this raises, nothing to roll back.

        try:

        except Exception:

            # payment-related exception; in single-threaded usage release
            # should succeed because we just reserved the same qty.
            pass
        raise
    return {"status": "confirmed", "qty": qty, "total": qty * price}
