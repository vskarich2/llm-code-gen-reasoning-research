def update_profile(user, changes):
    """Update user profile fields from changes dict.

    Invariant: display_name must always equal name.
    Behavior:
    - If 'name' is present in changes it is authoritative and both user['name'] and
      user['display_name'] will be set to that value.
    - Else if 'display_name' is present (and 'name' is not), both fields will be set
      to the provided display_name value so the invariant remains true.
    - Other fields (email, age) are updated normally.
    """
    # Determine canonical name source in a deterministic way.
    new_name = None
    if "name" in changes:
        new_name = changes["name"]
    elif "display_name" in changes:
        new_name = changes["display_name"]

    if new_name is not None:
        # Ensure both fields are kept in sync to maintain the invariant.
        user["name"] = new_name
        user["display_name"] = new_name

    # Apply other changes, skipping name/display_name since handled above.
    for key, value in changes.items():
        if key in ("name", "display_name"):
            continue
        if key == "email":
            user["email"] = value
        elif key == "age":
            user["age"] = value
    return user
