```python
"""Metrics pipeline: compute raw stats before transforming data.

This module computes simple statistics on the original data and produces a
normalized (0-1) version of the data. The pipeline guarantees that the
reported raw statistics always come from the original (untransformed) data.
"""


def compute_raw_stats(data):
    """Compute statistics on raw (untransformed) data.

    Returns a dict with keys 'raw_max', 'raw_min', and 'raw_sum'. For empty or
    falsy inputs the function returns zeros to keep the contract simple.
    """
    if not data:
        return {"raw_max": 0, "raw_min": 0, "raw_sum": 0}
    return {
        "raw_max": max(data),
        "raw_min": min(data),
        "raw_sum": sum(data),
    }


def normalize(data):
    """Normalize numeric data to the 0-1 range.

    - For empty inputs returns an empty list.
    - If all values are equal returns a list of 0.5 to indicate a centered value.
    """
    if not data:
        return []
    lo, hi = min(data), max(data)
    if hi == lo:
        return [0.5] * len(data)
    return [(x - lo) / (hi - lo) for x in data]


def pipeline(data):
    """Run pipeline: compute raw stats from original data, then normalize.

    raw_stats MUST reflect the original data, not the normalized version.
    """
    # Compute raw statistics from the original data first so they remain
    # independent of subsequent transformations.
    raw_stats = compute_raw_stats(data)
    cleaned = normalize(data)
    return {"raw_stats": raw_stats, "cleaned": cleaned}


def format_report(result):
    """Format pipeline result for display."""
    return f"max={result['raw_stats']['raw_max']}"

```
