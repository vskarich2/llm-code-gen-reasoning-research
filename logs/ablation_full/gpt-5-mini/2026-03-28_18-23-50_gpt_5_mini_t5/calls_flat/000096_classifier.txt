def send_with_retry(msg, max_retries=2, fail_first=False):
    """Retry send on transient errors.

    Raises the last ConnectionError if all retries fail so callers can
    observe failure and retry at a higher level if desired.
    """
    last_exc = None
    for attempt in range(max_retries):
        try:

            continue
    # All retries exhausted; propagate the last exception to the caller
    if last_exc is not None:
        raise last_exc
    # Defensive fallback (shouldn't happen)
    raise ConnectionError("send_with_retry: failed without exception")

    return _attempt_count
