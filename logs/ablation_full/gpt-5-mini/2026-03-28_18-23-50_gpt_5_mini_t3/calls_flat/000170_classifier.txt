```python
import re
import warnings

"""Feature flag system with snake_case keys.

This module accepts both snake_case and camelCase flag names in is_enabled by
normalizing inputs to snake_case. Unknown flags return False (preserving the
original behavior) but emit a warning so missing lookups are visible during
development.
"""

FLAGS = {
    "dark_mode": True,
    "beta_features": False,
    "new_dashboard": True,
    "analytics_v2": False,
}


def _to_snake(name):
    """Normalize a flag name to snake_case.

    Accepts camelCase, PascalCase, or already snake_case strings and returns
    a predictable snake_case representation. Raises TypeError for non-str
    inputs.
    """
    if not isinstance(name, str):
        raise TypeError(f"flag_name must be a str, got {type(name)!r}")
    if not name:
        return ""
    # Insert underscores between lower->Upper or number->Upper transitions,
    # then collapse sequences, finally lowercase.
    s1 = re.sub(r"(.)([A-Z][a-z]+)", r"\1_\2", name)
    s2 = re.sub(r"([a-z0-9])([A-Z])", r"\1_\2", s1)
    return s2.replace('-', '_').lower()


def is_enabled(flag_name):
    """Check if a feature flag is enabled.

    Args:
        flag_name: the flag key to look up. Can be snake_case or camelCase.

    Returns:
        True if the feature flag is enabled, False otherwise.

    Behavior notes:
    - Incoming names are normalized to snake_case before lookup (so callers can
      use camelCase like 'darkMode' and still find 'dark_mode').
    - If the normalized flag is not present in FLAGS, the function returns
      False (maintaining prior behavior) but emits a UserWarning to avoid
      silently masking configuration problems.
    """
    normalized = _to_snake(flag_name)

    if normalized in FLAGS:
        return FLAGS[normalized]

    warnings.warn(
        "Unknown feature flag '%s' (normalized: '%s'). Returning False." % (flag_name, normalized),
        UserWarning,
    )
    return False


def list_flags():
    """Return all flag names (snake_case) currently configured."""
    return list(FLAGS.keys())

```
