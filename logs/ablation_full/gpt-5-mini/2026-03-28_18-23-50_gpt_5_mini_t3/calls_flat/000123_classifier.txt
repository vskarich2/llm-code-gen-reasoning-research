"""Access control: map roles to permission sets."""

# Canonical, immutable permission sets per role (use lowercase role keys).
ROLE_PERMISSIONS = {
    "admin": frozenset({"read", "write", "delete", "manage_users"}),
    "user": frozenset({"read", "write"}),
    "moderator": frozenset({"read", "write", "delete"}),
}


def get_permissions(role, default=None):
    """
    Return a new mutable set of permissions for the given role.

    - Role lookup is case-insensitive (role is normalized with str(...).lower()).
    - The function returns a fresh set copy of the stored frozenset so callers
      can mutate the returned set without changing the global ROLE_PERMISSIONS.
    - If role is not found and a default is provided, a copy of default is
      returned. Otherwise an empty set is returned.
    """
    # Normalize role to allow callers using different casing (e.g. 'Moderator').
    key = "" if role is None else str(role).lower()
    perms = ROLE_PERMISSIONS.get(key)
    if perms is not None:
        # Return a mutable copy to callers, preserving the immutable canonical set.
        return set(perms)
    if default is not None:
        return set(default)
    return set()


def has_permission(role, action):
    """Check whether a role has a specific permission."""
    return action in get_permissions(role)
