from validation import validate_email, validate_name
from notifications import build_greeting, should_reverify


def create_user(name, email):
    """Create a user with verified status and cached greeting."""
    return {
        "name": name,
        "email": email,
        "verified": False,
        "cached_greeting": build_greeting({"name": name}),
    }


def verify_user(user):
    """Mark user as verified."""
    user["verified"] = True
    return user


def update_profile(user, changes):
    """Update profile fields.

    Invariant: changing email must set verified=False and clear cached_greeting.
    Invariant: changing name must update cached_greeting.
    """
    # Track what actually changed after validation so we can enforce invariants
    email_changed = False
    name_changed = False
    old_email = user.get("email")

    # First, apply validated field updates and record flags
    for key, value in changes.items():
        if key == "email" and validate_email(value):
            # apply email change, but defer side-effects until after we know
            # if name also changed (to resolve cached_greeting conflict)
            user["email"] = value
            email_changed = True
        elif key == "name" and validate_name(value):
            user["name"] = value
            name_changed = True

    # Post-process invariants deterministically
    if email_changed:
        # If the email actually changed in a meaningful way, require reverify
        if should_reverify(old_email, user.get("email")):
            user["verified"] = False
        # If name was not changed in the same update, clear the cached greeting
        if not name_changed:
            user["cached_greeting"] = ""

    if name_changed:
        # Name change always updates cached greeting to reflect the new name
        user["cached_greeting"] = build_greeting(user)

    return user
