def update_profile(user, changes):
    """Update user profile fields from changes dict.

    Invariant: display_name must always equal name.
    """
    # Handle name/display_name first to ensure the invariant is preserved.
    name_value = None
    if "name" in changes:
        name_value = changes["name"]
    elif "display_name" in changes:
        name_value = changes["display_name"]

    if name_value is not None:
        user["name"] = name_value
        user["display_name"] = name_value

    for key, value in changes.items():
        if key in ("name", "display_name"):
            # already handled above to keep name and display_name in sync
            continue
        elif key == "email":
            user["email"] = value
        elif key == "age":
            user["age"] = value
