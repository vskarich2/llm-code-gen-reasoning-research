_settings = {"host": "localhost", "port": 8080, "debug": False}


def get_host():
    """Return the current host setting."""
    # Read from the live settings mapping so we always observe updates
    return _settings["host"]


def get_settings():
    """Return a shallow copy of the current settings dict."""
    return dict(_settings)


def reset_settings():
    """Reset settings to default values (useful for testing)."""
    global _settings
    _settings = {"host": "localhost", "port": 8080, "debug": False}


def configure(host=None, port=None, debug=None):
    """Update settings in-place. Only non-None arguments are applied."""
    if host is not None:
        _settings["host"] = host
    if port is not None:
        _settings["port"] = port
    if debug is not None:
