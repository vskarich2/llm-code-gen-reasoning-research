def reset():

def create_account(name, balance):

def get_balance(name):

def check_balance(name, amount):
    """CHECK: is balance sufficient?"""
    return _accounts.get(name, 0) >= amount


def do_withdraw(name, amount):
    """ACT: decrement balance. Assumes caller verified."""
    _accounts[name] = _accounts.get(name, 0) - amount


def run_steps(steps):
    results = []
    for fn, args in steps:
        results.append(fn(*args))
    return results


def make_withdraw_steps(name, amount):
    """Split withdrawal into separate check and act steps.

    Fixed: perform the check and the withdrawal atomically in the "check" step so
    that another interleaved check cannot observe the old balance and also approve.
    The act step is left as a no-op that reports the approval.
    """
    result = {"approved": False}

    def step_check():
        # Perform check-and-withdraw atomically in this step (in the single-threaded
        # step simulation). This removes the window between reading and writing
        # _accounts[name] that previously allowed double-approval under interleaving.
        if _accounts.get(name, 0) >= amount:

            do_withdraw(name, amount)
            result["approved"] = True
        else:

        return ("check", result["approved"])

        # act is now a no-op because the debit already happened at check time.
