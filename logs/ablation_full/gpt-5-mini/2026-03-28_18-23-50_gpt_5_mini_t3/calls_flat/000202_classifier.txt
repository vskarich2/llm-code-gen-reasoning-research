def sync_user_to_cache(user, overwrite=True):
    """
    Consolidated helper for putting a user's display name into the cache.
    - If overwrite is True, always set the cache to user['name'] (cache_put).
    - If overwrite is False, only set if absent (cache_put_if_absent).
    The function is defensive: it returns without exception if user is malformed.
    """
    if not isinstance(user, dict):
        return
    uid = user.get("id")
    name = user.get("name")
    if uid is None or name is None:
        return
    key = f"user:{uid}"
    if overwrite:
        cache_put(key, name)
    else:
        cache_put_if_absent(key, name)
