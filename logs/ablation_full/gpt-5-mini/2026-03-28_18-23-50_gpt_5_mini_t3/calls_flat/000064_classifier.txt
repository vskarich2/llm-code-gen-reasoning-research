 """Batch processor with per-item snapshots."""

_counter = 0

_processed_count = 0


def reset():
    """Reset all internal state (counter, snapshots, processed count)."""
    global _counter, _snapshots, _processed_count
    _counter = 0
    _snapshots = []
    _processed_count = 0


def snapshot():
    """Record the current counter value.

    This appends the current _counter to _snapshots. It is a thin helper
    used by process_batch; external callers may call it but doing so will
    affect verify_consistency() since it counts snapshots produced by
    processing.
    """
    _snapshots.append(_counter)


def get_snapshots():
    """Return a shallow copy of the snapshots."""
    return list(_snapshots)


def process_batch(items):
    """Process each item: increment counter and take a snapshot after each.

    For each item in items we:
      - add the item's value to _counter
      - record a snapshot of the new counter value
      - increment the processed-item counter

    Returns the updated _counter.
    """
    global _counter, _processed_count
    for item in items:
        _counter += item
        snapshot()  # record one snapshot per processed item
        _processed_count += 1
    return _counter


def verify_consistency():
    """Verify that each processed item produced exactly one snapshot.

    Returns True when len(_snapshots) equals the tracked number of processed
    items, False otherwise.
    """
    return len(_snapshots) == _processed_count
