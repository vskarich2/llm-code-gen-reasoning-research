def add_product(product_id, name, price):

    # Keep cache consistent: store a copy of the DB entry
    _cache[product_id] = dict(_db[product_id])


def get_product(product_id):
    """Get product by ID, using cache for repeated lookups.

    Returns a fresh copy of the cached representation so callers cannot
    mutate internal cache state.
    """
    if product_id in _cache:
        # Return a copy to prevent external mutation from affecting the cache
        return dict(_cache[product_id])
    product = _db.get(product_id)
    if product is not None:
        # Cache a copy of the DB entry to keep cache and DB decoupled
        _cache[product_id] = dict(product)
        return dict(_cache[product_id])
    return None


def update_product(product_id, **fields):
    """Update product fields in the database and keep cache consistent.

    Invariant: get_product() must return current data after update.
    To preserve that invariant we update the cached representation (replace
    it with a fresh copy of the updated DB entry) when the database is
    successfully updated.
    """
    if product_id not in _db:
        return False
    _db[product_id].update(fields)
    # Ensure cache reflects the DB after the update to avoid stale reads.
    _cache[product_id] = dict(_db[product_id])
    return True

    """Clear all data."""
    _db.clear()
