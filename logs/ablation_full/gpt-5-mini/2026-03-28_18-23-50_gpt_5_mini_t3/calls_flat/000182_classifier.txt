def normalize(entries):
    seen = set()
    out = []
    for e in entries:
        k = e["id"]
        if k not in seen:
            seen.add(k)
            out.append(e)
    return out


def project(entries):
    return [{"id": e["id"], "label": e.get("label", ""), "val": e.get("val", 0)} for e in entries]


def collapse(entries):
    merged = {}
    for e in entries:
        k = e["id"]
        if k in merged:
            merged[k]["val"] = merged[k]["val"] + e.get("val", 0)
        else:
            merged[k] = dict(e)
    return list(merged.values())


def stage(st, processed, freeze=True):
    """
    Stage processed entries into pending and view. By default (freeze=True) also commit them into stable and mark meta.frozen=True.
    If freeze=False, only pending and view are updated (useful for preview).
    """
    st["pending"] = list(processed)
    st["view"] = project(processed)
    st["meta"]["version"] += 1
    if freeze:
        # finalize into stable and mark frozen to indicate committed state
        st["stable"] = list(processed)
        st["meta"]["frozen"] = True


def commit(st):
    st["stable"] = list(st["pending"])
    st["meta"]["frozen"] = True


def freeze_view(st):
    st["view"] = project(st["stable"])


def materialize(st):
    return {
        "items": list(st["stable"]),
        "display": list(st["view"]),
        "v": st["meta"]["version"],
    }


from state import make_state, copy_state
from reducers import normalize, collapse, stage, materialize, project


def process_batch(entries):
    st = make_state(entries)
    raw = st["raw"]

    # dedupe
    cleaned = normalize(raw)

    # merge values for same id
    merged = collapse(cleaned)

    # stage puts merged into pending + builds view from merged
    # stage now also finalizes into stable by default

    out = materialize(st)
    return st, out


def process_incremental(prev_state, new_entries):

    out = materialize(st)
    return st, out
