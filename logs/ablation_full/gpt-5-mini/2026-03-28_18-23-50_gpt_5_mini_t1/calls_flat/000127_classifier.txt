from roles import admin_access, user_access, moderator_access, guest_access


# Dispatch table mapping roles to access handlers
_ROLE_DISPATCH = {
    "admin": admin_access,
    "user": user_access,
    "moderator": moderator_access,
    "guest": guest_access,
}


def _default_access():
    """Default: no access for unknown roles."""
    return {"read": False, "write": False, "delete": False}


def get_access(role):
    """Return the access dict for a given role.

    Normalize role strings (strip + lower) so callers using different case or
    surrounding whitespace still match the dispatch table. Non-string roles are
    looked up as-is (and will fallback to default if not present).
    """
    key = role.strip().lower() if isinstance(role, str) else role
    handler = _ROLE_DISPATCH.get(key, _default_access)
    return handler()


def can_read(role):
    """Check if role has read access."""
    return get_access(role).get("read", False)
