from billing import create_invoice, get_invoices
from flags import enable, disable, is_enabled


def checkout(customer, items, use_new_pricing=False):
    # Save previous state so we can restore it after invoice creation
    prev = is_enabled("new_pricing")

    # Set the flag according to the caller's request
    if use_new_pricing:
        enable("new_pricing")
    else:
        disable("new_pricing")

    try:
        invoice = create_invoice(customer, items)
    finally:
        # Restore previous state to avoid leaking flag changes
        if prev:
            enable("new_pricing")
        else:
            disable("new_pricing")

    return invoice
