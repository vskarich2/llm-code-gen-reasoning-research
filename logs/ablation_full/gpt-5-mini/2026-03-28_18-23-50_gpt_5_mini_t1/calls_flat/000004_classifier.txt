import copy

DEFAULTS = {"timeout": 30, "retries": 3, "debug": False}


def create_config(overrides=None):
    """Return a config dict based on defaults, with optional overrides.

    This returns a fresh deep copy of DEFAULTS and applies any provided
    overrides so callers cannot accidentally mutate the global DEFAULTS.
    """
    base = copy.deepcopy(DEFAULTS)
    if overrides:
        if not isinstance(overrides, dict):
            raise TypeError("overrides must be a dict or None")
        base.update(overrides)
