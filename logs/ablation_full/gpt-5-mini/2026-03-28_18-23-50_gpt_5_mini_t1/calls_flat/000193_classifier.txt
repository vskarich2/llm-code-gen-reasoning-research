from models import Account
from ledger import record_debit, record_credit, record_transfer_attempt
from audit import emit_transfer_event, emit_failure_alert


def validate_transfer(sender, amount):
    if amount <= 0:
        raise ValueError("amount must be positive")
    if sender.balance < amount:
        raise ValueError("insufficient funds")


def _apply_transfer(sender, receiver, amount):
    """
    Core atomic transfer: performs the balance mutations and guarantees rollback
    if an exception occurs during the operation.
    This helper does NOT perform any observability (ledger/audit) writes.
    """
    validate_transfer(sender, amount)

    # Apply debit first
    sender.balance -= amount
    try:
        # Simulate a transient failure that can occur during the credit phase.
        # If it fires, ensure we rollback the debit before propagating.
        if random.random() < 0.3:
            raise RuntimeError("transient failure during credit")

        # Apply credit after the transient failure point. If any exception is
        # raised between debit and credit, the except block will restore sender.
        receiver.balance += amount

    except Exception:
        # Rollback the debit to keep balances consistent on failure.
        sender.balance += amount
        raise


def execute_transfer(sender, receiver, amount):
    # Record that an attempt started (pending). We will append a final status
    # entry after we know success/failure.
    record_transfer_attempt(sender.account_id, receiver.account_id, amount, status="pending")

    try:
        _apply_transfer(sender, receiver, amount)
    except Exception as exc:
        # Mark attempt as failed and emit a failure alert. No debit/credit
        # ledger entries are created because the core transfer rolled back.
        record_transfer_attempt(sender.account_id, receiver.account_id, amount, status="failed")
        emit_failure_alert(sender.account_id, receiver.account_id, amount, str(exc))
        raise
    else:
        # Core succeeded: record ledger entries and emit success event.
        record_debit(sender.account_id, amount)
        record_credit(receiver.account_id, amount)
        record_transfer_attempt(sender.account_id, receiver.account_id, amount, status="success")
        emit_transfer_event(sender.account_id, receiver.account_id, amount)
