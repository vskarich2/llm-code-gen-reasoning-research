class Report:
    """A simple table with column headers and data rows."""

    def __init__(self, headers):
        self.headers = list(headers)
        self.rows = []

    def add_row(self, *values):
        """Add a row of data."""
        row = make_row(*values)
        # enforce invariant: every row must have exactly as many columns as headers
        if not validate_row(row, self.column_count()):
            raise ValueError(f"Row has {len(row)} columns but expected {self.column_count()} per headers")
        self.rows.append(row)

    def delete_column(self, index):
        """Remove a column by index from headers and all rows.

        Must remove the header AND the corresponding element
        from every row to keep them aligned.
        """
        expected = len(self.headers)
        # validate index against current headers length (support negative indices)
        if not (-expected <= index < expected):
            raise IndexError("Column index out of range")
        # normalize index to non-negative for row operations
        idx = index if index >= 0 else expected + index
        # ensure rows are consistent with headers before mutating
        for r in self.rows:
            if len(r) != expected:
                raise ValueError(f"Inconsistent row length: {len(r)} != headers {expected}")
        # remove header and corresponding element from each row
        del self.headers[idx]
        for r in self.rows:
            del r[idx]

    def render(self):
        """Render the table as a list of dicts keyed by header."""
        result = []
        for row in self.rows:
            result.append(dict(zip(self.headers, row)))
        return result

    def get_column(self, index):
        """Return all values in a given column."""
        return [row[index] for row in self.rows]

    def column_count(self):
        """Return the number of columns based on headers."""
        return len(self.headers)
