 """Order fulfillment with inventory and wallet."""


class Inventory:
    def __init__(self, stock):
        self.stock = stock
        self.reserved = 0

    def reserve(self, qty):
        if qty > self.stock - self.reserved:
            raise ValueError("insufficient stock")
        self.reserved += qty

    def release(self, qty):
        self.reserved -= qty

    def available(self):
        return self.stock - self.reserved


class Wallet:
    def __init__(self, balance):
        self.balance = balance

    def charge(self, amount):
        if amount > self.balance:
            raise ValueError("insufficient funds")
        self.balance -= amount


def place_order(inventory, wallet, qty, price):
    """Reserve inventory, then charge wallet. Release reservation on any failure.

    Behavior:
    - Reserve inventory before attempting payment.
    - If payment fails (raises any exception), release the reserved quantity
      to avoid leaking reservations and re-raise the original exception.
    """
    # Perform reservation first; if this raises, nothing was reserved and no
    # release is necessary.
    inventory.reserve(qty)
    try:
        # Attempt to charge the wallet. If this raises for any reason we must
        # release the previously made reservation to maintain invariants.
        wallet.charge(qty * price)
    except BaseException:
        # Ensure reservation is released on *any* failure (including
        # KeyboardInterrupt/SystemExit) so we do not leave stock locked.
        inventory.release(qty)
        raise

    return {"status": "confirmed", "qty": qty, "total": qty * price}
