"""Simple rate limiter utilities.

This module provides small helper functions to check and update a request
count against a maximum allowed limit.
"""

from typing import Tuple


def is_rate_limited(count: int, limit: int) -> bool:
    """Return True if 'count' has reached or exceeded 'limit'.

    Args:
        count: number of requests already made (must be >= 0).
        limit: maximum allowed requests (must be >= 0).

    Returns:
        True if the caller should be blocked (count >= limit), False otherwise.

    Example:
        If limit == 3, counts 0, 1, 2 are allowed (is_rate_limited == False);
        counts 3 and above are blocked (is_rate_limited == True).
    """
    if count < 0 or limit < 0:
        raise ValueError("count and limit must be non-negative integers")
    return count >= limit


def check_and_increment(current_count: int, limit: int) -> Tuple[bool, int]:
    """Check rate limit and return (blocked, new_count).

    If not blocked, returns (False, current_count + 1). If blocked, returns
    (True, current_count) and does not increment.

    Args:
        current_count: current number of requests already made
        limit: maximum allowed requests

    Returns:
        A tuple (blocked, new_count).
    """
    if is_rate_limited(current_count, limit):
        return True, current_count
    return False, current_count + 1
