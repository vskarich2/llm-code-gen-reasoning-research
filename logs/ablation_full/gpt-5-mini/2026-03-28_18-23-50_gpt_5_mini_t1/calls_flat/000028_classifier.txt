_db = {}
_cache = {}


def add_product(product_id, name, price):
    """Add a product to the database and populate cache for immediate consistency."""
    _db[product_id] = {"name": name, "price": price}
    # Keep cache consistent with DB on add so immediate reads see the same snapshot
    _cache[product_id] = dict(_db[product_id])


def get_product(product_id):
    """Get product by ID, using cache for repeated lookups.

    Returns a shallow copy of the cached snapshot (or None) so callers cannot mutate
    the internal cached object.
    """
    if product_id in _cache:
        # return a copy to avoid external mutation of the cached dict
        return dict(_cache[product_id])
    product = _db.get(product_id)
    if product is not None:
        # cache a snapshot copy of the DB entry
        _cache[product_id] = dict(product)
        return dict(_cache[product_id])
    return None


def update_product(product_id, **fields):
    """Update product fields in the database.

    Invariant: get_product() must return current data after update.
    This function now updates the cache snapshot after modifying the DB so the
    invariant holds in single-threaded usage.
    Returns True when an existing product is updated, False if product_id not found.
    """
    if product_id not in _db:
        return False
    # mutate the DB record
    _db[product_id].update(fields)
    # Synchronize cache to reflect the updated DB snapshot (prevents stale reads)
    _cache[product_id] = dict(_db[product_id])
    return True


def reset():
    """Clear all data."""
    _db.clear()
    _cache.clear()
