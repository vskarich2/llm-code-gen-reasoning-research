import re

"""Feature flag system with snake_case keys.

This module normalizes common camelCase inputs to snake_case and raises
an explicit KeyError for unknown flags to avoid silent false negatives.
"""

FLAGS = {
    "dark_mode": True,
    "beta_features": False,
    "new_dashboard": True,
    "analytics_v2": False,
}


def _to_snake_case(name):
    """Convert camelCase or mixedCase to snake_case.

    - Preserves existing snake_case.
    - Replaces spaces and hyphens with underscores.
    - Lowercases the result.
    """
    if not isinstance(name, str):
        raise TypeError("flag name must be a string")
    # turn 'HTMLParser' -> 'html_parser', 'darkMode' -> 'dark_mode'
    s1 = re.sub(r'(.)([A-Z][a-z]+)', r'\1_\2', name)
    s2 = re.sub(r'([a-z0-9])([A-Z])', r'\1_\2', s1)
    s3 = s2.replace('-', '_').replace(' ', '_')
    return s3.lower()


def is_enabled(flag_name):
    """Check if a feature flag is enabled.

    Accepts either the canonical snake_case key or a camelCase equivalent.
    Raises KeyError for unknown/typoed flag names to avoid silent failures.

    Args:
        flag_name: the flag key to look up (snake_case or camelCase)

    Returns:
        True if the flag is enabled, False if disabled.

    Raises:
        TypeError: if flag_name is not a string.
        KeyError: if the flag is unknown (to avoid silent defaults).
    """
    if not isinstance(flag_name, str):
        raise TypeError("flag name must be a string")

    # Direct match (preferred)
    if flag_name in FLAGS:
        return bool(FLAGS[flag_name])

    # Try normalizing common camelCase -> snake_case
    normalized = _to_snake_case(flag_name)
    if normalized in FLAGS:
        return bool(FLAGS[normalized])

    # Explicit failure instead of silent default to expose typos/mismatches
    available = ", ".join(sorted(FLAGS.keys()))
    raise KeyError(f"Unknown feature flag '{flag_name}'. Known flags: {available}")


def list_flags():
    """Return all flag names (canonical snake_case keys)."""
    return list(FLAGS.keys())
