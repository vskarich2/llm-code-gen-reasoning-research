For example, if 'schedule_one' is called twice and 'schedule_batch' is called once, the history will incorrectly reflect three calls to 'schedule_one' instead of two, leading to inaccurate statistics.

## Broken Invariant

This is achieved by changing the default value of 'history' to 'None' and initializing it to an empty list within the function.

from worker import process, batch_process


def with_history(func, history=None):
    """Decorator that records call history for a function.

    Invariant: each decorated function must have its OWN
    independent history list.
    """
    # FIX: history default is now None, so each function gets its own list.
    if history is None:
        history = []

    def wrapper(*args, **kwargs):
        result = func(*args, **kwargs)
        history.append({"func": func.__name__, "args_count": len(args)})
        return result

    wrapper.get_history = lambda: list(history)
    wrapper.clear_history = lambda: history.clear()
    return wrapper


@with_history
def schedule_one(task):
    """Schedule and process a single task."""
    return process(task)


@with_history
def schedule_batch(tasks):
    """Schedule and process a batch of tasks."""
    return batch_process(tasks)
