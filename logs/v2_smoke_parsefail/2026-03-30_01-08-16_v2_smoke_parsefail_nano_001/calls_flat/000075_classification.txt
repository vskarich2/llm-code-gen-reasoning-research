_invoices = []

def create_invoice(customer, items, use_new_pricing=False):
    total = 0
    lines = []
    for item in items:
        price = compute_price(item["base"], item["qty"])
        lines.append({"sku": item["sku"], "amount": price})
        total += price

    invoice = {
        "customer": customer,
        "lines": lines,
        "total": total,
        "pricing_version": "v2" if use_new_pricing else "v1",
    }

    if is_enabled("audit_mode"):
        invoice["reasoning_evaluator_audit"] = True
        _invoices.append(invoice)

    return invoice
