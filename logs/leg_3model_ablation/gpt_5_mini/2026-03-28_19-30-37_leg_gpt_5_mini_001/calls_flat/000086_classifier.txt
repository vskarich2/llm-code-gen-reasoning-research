"""Feature flag system with snake_case keys.

Behavior changes:
- Accept camelCase/PascalCase flag names by normalizing to snake_case.
- Fail fast for unknown flags by raising KeyError instead of silently returning False.
"""

FLAGS = {
    "dark_mode": True,
    "beta_features": False,
    "new_dashboard": True,
    "analytics_v2": False,
}


def _to_snake_case(name):
    """Convert camelCase or PascalCase (and simple mixed cases) to snake_case.

    Examples:
        'darkMode' -> 'dark_mode'
        'DarkMode' -> 'dark_mode'
        'analyticsV2' -> 'analytics_v2'
    """
    if not isinstance(name, str):
        raise TypeError("flag name must be a string")
    # Insert underscore between lower-to-upper and between letter-to-digit transitions
    s1 = re.sub("(.)([A-Z][a-z]+)", r"\1_\2", name)
    s2 = re.sub("([a-z0-9])([A-Z])", r"\1_\2", s1)
    # Replace non-alphanumeric characters with underscore, then lowercase
    s3 = re.sub("[^0-9a-zA-Z]+", "_", s2)
    return s3.lower()


def is_enabled(flag_name):
