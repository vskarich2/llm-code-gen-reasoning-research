import copy

DEFAULTS = {"timeout": 30, "retries": 3, "debug": False}


def create_config(overrides=None):
    """Return a config dict based on defaults, with optional overrides.

    The returned dictionary is a deep copy of the module-level DEFAULTS so
    callers may modify it without altering the shared DEFAULTS object.
    If overrides is provided it must be a mapping and its keys will replace

    """

    if overrides:
        if not isinstance(overrides, dict):
