```python
"""Product catalog with caching.

This module stores products in an in-memory _db and keeps a simple _cache
for repeated lookups. The cache is synchronized on updates to ensure
get_product always returns current data.
"""
import threading

_db = {}
_cache = {}
_lock = threading.RLock()


def add_product(product_id, name, price):
    """Add a product to the database."""
    with _lock:
        _db[product_id] = {"name": name, "price": price}


def get_product(product_id):
    """Get product by ID, using cache for repeated lookups.

    Returns a shallow copy of the cached product dict (or None).
    """
    with _lock:
        if product_id in _cache:
            # Return a copy so callers cannot mutate the cached object
            return dict(_cache[product_id])
        product = _db.get(product_id)
        if product is not None:
            # Store a copy in cache and return a copy to the caller
            _cache[product_id] = dict(product)
            return dict(_cache[product_id])
        return None


def update_product(product_id, **fields):
    """Update product fields in the database.

    Invariant: get_product() must return current data after update.
    """
    with _lock:
        if product_id not in _db:
            return False
        _db[product_id].update(fields)
        # Keep cache synchronized: replace cached snapshot with updated data
        if product_id in _cache:
            _cache[product_id] = dict(_db[product_id])
        return True


def reset():
    """Clear all data."""
    with _lock:
        _db.clear()
        _cache.clear()

```
