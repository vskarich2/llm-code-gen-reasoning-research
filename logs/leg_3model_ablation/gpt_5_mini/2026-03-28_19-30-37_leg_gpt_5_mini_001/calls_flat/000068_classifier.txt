```python
 """Simple rate limiter.

This module provides a tiny utility to decide whether a caller should be
blocked based on how many requests have already been recorded.

Semantics:
- count is the number of requests already made (an integer >= 0).
- limit is the maximum allowed requests (an integer >= 0).

When count >= limit the caller must be blocked. Otherwise the caller is
allowed and the recorded count should be incremented.
"""

from typing import Tuple


def is_rate_limited(count: int, limit: int) -> bool:
    """Return True if the request count has reached or exceeded the rate limit.

    Args:
        count: current number of requests made
        limit: maximum allowed requests

    Returns:
        True if the caller should be blocked, False if allowed.
    """
    # Block when the recorded count is greater than or equal to the limit.
    return count >= limit


def check_and_increment(current_count: int, limit: int) -> Tuple[bool, int]:
    """Check rate limit and return (blocked, new_count).

    If not blocked, the count is incremented and the caller is allowed.
    """
    if is_rate_limited(current_count, limit):
        return True, current_count
    return False, current_count + 1

```
