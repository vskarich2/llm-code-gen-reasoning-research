def run_steps(steps):

    for fn, args in steps:

    return results


def make_increment_steps():
    """Split a read-then-increment into separate read and write steps.

    BUG: under interleaving, two increments both read the same value
    and both write value+1, losing one update.
    """

        # Re-read the current value at the time of the write so that

        # if another increment wrote in between the read and write steps,
        # we don't overwrite that update with a stale increment.
        new_value = get() + 1
        _set(new_value)
        return ("write", new_value)

    return step_read, step_write
