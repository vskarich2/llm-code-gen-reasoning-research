'''Scheduler with decorator that tracks call history.'''

from worker import process, batch_process

_shared_log = []

def with_history(func):
    """Decorator that records call history for a function."""
    def wrapper(*args, **kwargs):
        # Initialize a new history list for each decorated function
        if not hasattr(wrapper, '_history'):
            wrapper._history = []
        result = func(*args, **kwargs)
        wrapper._history.append({"func": func.__name__, "args_count": len(args)})
        return result
    # Provide methods to access and clear history
    wrapper.get_history = lambda: list(getattr(wrapper, '_history', []))
    wrapper.clear_history = lambda: getattr(wrapper, '_history', []).clear()
    return wrapper

@with_history
def schedule_one(task):
    """Schedule and process a single task."""
    return process(task)

@with_history
def schedule_batch(tasks):
    """Schedule and process a batch of tasks."""
    return batch_process(tasks)


def get_all_stats():
    """Get combined stats (distractor)."""
    return {
        "one_calls": len(schedule_one.get_history()),
