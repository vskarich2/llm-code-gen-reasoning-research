def reset():

    _value = 0


def get():

def _set(v):

    _value = v


def run_steps(steps):
    results = []
    for fn, args in steps:
        results.append(fn(*args))
    return results


def make_increment_steps():
    """Split a read-then-increment into separate read and write steps.

    BUG: under interleaving, two increments both read the same value
    and both write value+1, losing one update.
    """

    def step_read():
        return get()

    def step_write(v):
        _set(v)
        return v

    def increment():
        current = get()
        new_value = current + 1
        _set(new_value)
        return new_value

    def make_steps():
        def step():
            current = get()
            new_value = current + 1
            _set(new_value)
            return new_value
        return step

def sequential_double_increment():

    reset()
